NEW
  PREfiX
AUTO 4,1
#!/bin/sh
#
#
#
echo -e "\f\n              A2osX Terminal Tests"
echo
echo "This tests the various Terminal/Console/Screen attributes"
echo "Available in A2osX scripts and applications."
if [ -N $log ]
  echo "Begin A2osX Terminal Tests" >> ${log}
else
  set T = 0
  set F = 0
  set P = 0
fi
set T = $T + 1
echo -e -N "\e[10;10H"
echo "This Text Should be on line 10 column 10"
echo
echo -e -N "\e[20;5H"
read -P "Does the line above appear to be in the right location (Y/N)?" R
echo
if [ $R = "y" ]
  set R = "Y"
fi
echo -e "\f"
echo -e -N "\e[9;64H"
if [ $R = "Y" ]
  echo "Test 1 PASSED"
  set P = $P + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 1 PASSED" >> ${log}
  fi
else
  set F = $F + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 1 FAILED" >> ${log}
  fi
  echo "Test 1 FAILED"
fi
#
#  Draw Box in middle of screen
#
#  Screen is 80 x 24, make box 40 x 10 centered so
#  starting at 7 - 20
#
set T = $T + 1
echo -e -N "\e[7;20H"
echo -e "\e[7m                                        \e[0m"
set LINE = 8
WHILE [ $LINE -NE 17 ]
  echo -e -N "\e[${LINE};20H"
  echo -e "\e[7m \e[0m                                      \e[7m \e[0m"
  set LINE = $LINE + 1
LOOP
echo -e -N "\e[17;20H"
echo -e "\e[7m                                        \e[0m"
echo
echo -e -N "\e[20;5H\e[K"
read -P "Is there a box centered on the screen above (Y/N)?" R
echo
if [ $R = "y" ]
  set R = "Y"
fi
echo -e -N "\e[10;64H"
if [ $R = "Y" ]
  echo "Test 2 PASSED"
  set P = $P + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 2 PASSED" >> ${log}
  fi
else
  set F = $F + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 2 FAILED" >> ${log}
  fi
  echo "Test 2 FAILED"
fi
#
#  Now put something in center of box
#
set T = $T + 1
echo -e -N "\e[12;30HA2osX is the Greatest!"
echo -e -N "\e[20;5H\e[K"
read -P "Did a phrase appear centered in the box (Y/N)?" R
echo
if [ $R = "y" ]
  set R = "Y"
fi
echo -e -N "\e[11;64H"
if [ $R = "Y" ]
  echo "Test 3 PASSED"
  set P = $P + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 3 PASSED" >> ${log}
  fi
else
  set F = $F + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 3 FAILED" >> ${log}
  fi
  echo "Test 3 FAILED"
fi
#
#  Now inverse the phrase in the box  \e[7m \e[0m
#
set T = $T + 1
echo -e -N "\e[12;30H\e[7mA2osX is the Greatest!\e[0m"
echo -e -N "\e[20;5H\e[K"
read -P "Is the phrase now in inverse (Y/N)?" R
echo
if [ $R = "y" ]
  set R = "Y"
fi
echo -e -N "\e[12;64H"
if [ $R = "Y" ]
  echo "Test 4 PASSED"
  set P = $P + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 4 PASSED" >> ${log}
  fi
else
  set F = $F + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 4 FAILED" >> ${log}
  fi
  echo "Test 4 FAILED"
fi
#
#  Put up full line Title Bar above box
#
set T = $T + 1
echo -e -N "\e[2;1H"
echo -N "------------------------------  Full Width Title  ------------------------------"
echo -e -N "\e[20;5H\e[K"
read -P "Has a title line appeared on line 2 (Y/N)?" R
echo
if [ $R = "y" ]
  set R = "Y"
fi
echo -e -N "\e[13;64H"
if [ $R = "Y" ]
  echo "Test 5 PASSED"
  set P = $P + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 5 PASSED" >> ${log}
  fi
else
  set F = $F + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 5 FAILED" >> ${log}
  fi
  echo "Test 5 FAILED"
fi
#
# Now Blank right half of line
#
set T = $T + 1
echo -e -N "\e[2;41H\e[K"
echo -e -N "\e[20;5H\e[K"
read -P "Did the right half of the title disappear (Y/N)?" R
echo
if [ $R = "y" ]
  set R = "Y"
fi
echo -e -N "\e[14;64H"
if [ $R = "Y" ]
  echo "Test 6 PASSED"
  set P = $P + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 6 PASSED" >> ${log}
  fi
else
  set F = $F + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 6 FAILED" >> ${log}
  fi
  echo "Test 6 FAILED"
fi
#
#  Put up full line Title Bar above box
#
echo -e -N "\e[2;1H"
echo -N "------------------------------  Full Width Title  ------------------------------"
echo -e -N "\e[20;5H\e[K"
read -P "Has a title line appeared on line 2 (Y/N)?" R
echo
#
# Now Blank left half of line
#
set T = $T + 1
echo -e -N "\e[2;40H\e[1K"
echo -e -N "\e[20;5H\e[K"
read -P "Did the right half of the title disappear (Y/N)?" R
echo
if [ $R = "y" ]
  set R = "Y"
fi
echo -e -N "\e[15;64H"
if [ $R = "Y" ]
  echo "Test 7 PASSED"
  set P = $P + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 7 PASSED" >> ${log}
  fi
else
  set F = $F + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 7 FAILED" >> ${log}
  fi
  echo "Test 7 FAILED"
fi
#
#  Put up full line Title Bar above box
#
echo -e -N "\e[2;1H"
echo -N "------------------------------  Full Width Title  ------------------------------"
echo -e -N "\e[20;5H\e[K"
read -P "Has a title line appeared on line 2 (Y/N)?" R
echo
#
# Now Blank entire line
#
set T = $T + 1
echo -e -N "\e[2;40H\e[2K"
echo -e -N "\e[20;5H\e[K"
read -P "Did the entire title line disappear (Y/N)?" R
echo
if [ $R = "y" ]
  set R = "Y"
fi
echo -e -N "\e[16;64H"
if [ $R = "Y" ]
  echo "Test 8 PASSED"
  set P = $P + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 8 PASSED" >> ${log}
  fi
else
  set F = $F + 1
  if [ -N $log ]
    echo "A2osX Terminal Test 8 FAILED" >> ${log}
  fi
  echo "Test 8 FAILED"
fi
#  End of Tests
if [ -N $log ]
  echo "End A2osX Terminal Tests" >> ${log}
fi
echo -e -N "\e[20;5H\e[K"
echo -e "\e[20;15HEnd of Tests"
echo
MAN
TEXT /MAKE/usr/share/tests/vttest
