NEW
  PREFIX 
AUTO 4,1
#!/bin/sh
#
#  This is the master script that calls all test scripts.
#  It keeps a count of passed/failed tests and writes results
#  to a log file.
#
#  First Clear the Screen
echo -e "\f"
#
echo -e "\n\n                          Build Test Script\n\n"
#    
#  Now get the build Number
#
read -P "            Enter Build Number: " B
echo
echo -e "\n\n   Running All Tests for Build: $B"
#
#  Check to see if VAR then VAR/log exits, if not create
#
if [ -D ${ROOT}var ]
else
    md ${ROOT}var
fi
if [ -D ${ROOT}var/log ]
else
    md ${ROOT}var/log
fi
if [ -D ${ROOT}var/log/tests ]
else
    md ${ROOT}var/log/tests
fi
#  Create log File for tests var/log/tests/BUILD.BuildNo
#
#  Write Header To log
#
set log = "${ROOT}var/log/tests/BUILD.${B}"
echo > ${log}
echo >> ${log}
echo "              Build Test Suite" >> ${log}
echo -e "\nBuild No: ${B}" >> ${log}
echo -N "    Date: " >> ${log}
DATE >> ${log}
echo -e "----------------------------------------\n" >> ${log}
#
#  Capture Mem to log
#
mem >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
#
#  First Create and set the Test Counters
#
set T = 1
set P = 0
set F = 0
#
#  Run The Tests using the current shell (. test)
#
. badshtest
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           Calling echoTEST\n" >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
. echotest
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           Calling VTTEST\n" >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
. vttest
#echo -e "\n----------------------------------------\n" >> ${log}
#echo -e "\n           Calling CATTEST\n" >> ${log}
#echo -e "\n----------------------------------------\n" >> ${log}
#. cattest
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           Calling ARGTEST\n" >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
. argtest
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           Calling ifLEVEL\n" >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
. iflevel
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           Calling MD5TEST\n" >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
. md5test
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           Calling ifZNTEST\n" >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
. ifzntest
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           Calling ifFILETEST\n" >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
. iffiletest
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           Calling ifSTRTEST\n" >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
. ifstrtest
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           Calling ifNUMTEST\n" >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
. ifnumtest
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           Calling ifANDORTEST\n" >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
. ifandortest
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           Calling WHILETEST\n" >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
. whiletest
#
#  End of Tests
#
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           All Tests Run\n" >> ${log}
echo "----------------------------------------"
echo "      Total Tests: ${T}"
echo "     Tests Passed: ${P}"
echo "     Tests Failed: ${F}"
echo -e "----------------------------------------\n"
#echo
echo -e "\n----------------------------------------" >> ${log}
echo "      Total Tests: ${T}" >> ${log}
echo "     Tests Passed: ${P}" >> ${log}
echo "     Tests Failed: ${F}" >> ${log}
echo "----------------------------------------" >> ${log}
echo
echo "Here are the variables that are still set" >> ${log}
echo "----------------------------------------" >> ${log}
echo >> ${log}
set >> ${log}
echo "----------------------------------------" >> ${log}
set T =
set P =
set F =
set B =
set log =
MAN
TEXT /MAKE/usr/share/tests/buildtest
