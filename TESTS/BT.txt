NEW
  PREFIX 
AUTO 4,1
#!/bin/sh
#
#  This is the master script that calls all test scripts.
#  It keeps a count of passed/failed tests and writes results
#  to a log file.
#
#  First Clear the Screen
echo -e "\f"
#
echo -e "\n\n                          Build Test Script\n\n"

#    
#  Now get the build Number
#
read -P "            Enter Build Number: " B
echo
echo -e "\n\n   Running All Tests for Build: $B"

#  Create log File for tests var/log/tests/BUILD.BuildNo
#
#  Write Header To log
#
md ${ROOT}var/log/tests
set log = "${ROOT}var/log/tests/BUILD.${B}"
echo > ${log}
echo >> ${log}
echo "              Build Test Suite" >> ${log}
echo -e "\nBuild No: ${B}" >> ${log}
echo -N "    Date: " >> ${log}
date >> ${log}
#echo -N "    Time: " >> ${log}
#time >> ${log}
echo -e "----------------------------------------\n" >> ${log}
#
#  Capture Mem to log
#
mem >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
#
#  First Create and set the Test Counters
#
set T = 1
set P = 0
set F = 0
#
#  Run The Tests using the current shell (. test)
#
. badshtest
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           Calling vttest\n" >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
. vttest
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           Calling argtest\n" >> ${log}
echo -e "\n----------------------------------------\n" >> ${log}
. argtest
echo -e "\n----------------------------------------\n" >> ${log}
#
#  End of Tests
#
echo -e "\n----------------------------------------\n" >> ${log}
echo -e "\n           All Tests Run\n" >> ${log}
echo "----------------------------------------"
echo "      Total Tests: ${T}"
echo "     Tests Passed: ${P}"
echo "     Tests Failed: ${F}"
echo -e "----------------------------------------\n"
#echo
echo -e "\n----------------------------------------" >> ${log}
echo "      Total Tests: ${T}" >> ${log}
echo "     Tests Passed: ${P}" >> ${log}
echo "     Tests Failed: ${F}" >> ${log}
echo "----------------------------------------" >> ${log}
echo
echo "Here are the variables that are still set" >> ${log}
echo "----------------------------------------" >> ${log}
echo >> ${log}
set >> ${log}
echo "----------------------------------------" >> ${log}
set T =
set P =
set F =
set B =
set log =
MAN
TEXT /MAKE/usr/share/tests/bt
