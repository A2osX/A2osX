AUTO 4,1
#!/bin/sh
echo -e "\f\nThis Script will perform a series of tests on the echo command"
echo -e "\nSince these are visual tests, you will be asked to verify that"
echo "each performed as expected by Responding (Y) or (N) after each test."
echo "Responses can be in UPPER or Lower case and are a single letter,"
echo "Y for Yes and N for No.  Return (blank or empty) will mean NO."
echo -e "\n\n"
read -P "Press Return to start tests" Z
#Clear Screen Test
if [ -Z $T ]
  set T = 0
fi
set T = $T + 1
echo -e \f
echo -e \n
read -P "Did the screen clear: " Z
if [ $Z = "y" ]
  set Z = "Y"
fi
if [ $Z = "Y" ]
  if [ -N $log ]
    set P = $P + 1
    echo "echo Clear Screen Passed" >> ${log}
  fi
else
  if [ -N $log ]
    set F = $F + 1
    echo "echo Clear Screen Failed" >> ${log}
  fi
fi
#Backspace Test
set T = $T + 1
echo
echo -e "\n"
echo -e "----> 1234\b\b5678 <----"
echo -e "\n"
read -P "Is 125678 displayed between the arrows: " Z
if [ $Z = "y" ]
  set Z = "Y"
fi
if [ $Z = "Y" ]
  if [ -N $log ]
    set P = $P + 1
    echo "echo Backspace Passed" >> ${log}
  fi
else
  if [ -N $log ]
    set F = $F + 1
    echo "echo Backspace Failed" >> ${log}
  fi
fi
#Inverse Test
set T = $T + 1
echo
echo -e "\n"
echo -e -e "----> abc \e[7mdef\e[0m ghi <----"
echo -e "\n"
read -P "Is def above inversed: " Z
if [ $Z = "y" ]
  set Z = "Y"
fi
if [ $Z = "Y" ]
  if [ -N $log ]
    set P = $P + 1
    echo "echo Inverse Passed" >> ${log}
  fi
else
  if [ -N $log ]
    set F = $F + 1
    echo "echo Inverse Failed" >> ${log}
  fi
fi
#Newline Test
set T = $T + 1
echo
echo -e "\n"
echo -e "----> abc\ndef <----"
echo -e "\n"
read -P "Does abcdef appear split between two lines: " Z
if [ $Z = "y" ]
  set Z = "Y"
fi
if [ $Z = "Y" ]
  if [ -N $log ]
    set P = $P + 1
    echo "echo Newline Passed" >> ${log}
  fi
else
  if [ -N $log ]
    set F = $F + 1
    echo "echo Newline Failed" >> ${log}
  fi
fi
#Backslash Test
set T = $T + 1
echo
echo -e "\n"
echo -e "----> \\ <----"
echo -e "\n"
read -P "Is there a blackslash above: " Z
if [ $Z = "y" ]
  set Z = "Y"
fi
if [ $Z = "Y" ]
  if [ -N $log ]
    set P = $P + 1
    echo "echo Backslash Passed" >> ${log}
  fi
else
  if [ -N $log ]
    set F = $F + 1
    echo "echo Backslash Failed" >> ${log}
  fi
fi
#Percent Test
set T = $T + 1
echo
echo -e "\n"
echo -e "----> \% <----"
echo -e "\n"
read -P "Is there a percent symbol above: " Z
if [ $Z = "y" ]
  set Z = "Y"
fi
if [ $Z = "Y" ]
  if [ -N $log ]
    set P = $P + 1
    echo "echo Percent Passed" >> ${log}
  fi
else
  if [ -N $log ]
    set F = $F + 1
    echo "echo Percent Failed" >> ${log}
  fi
fi
#Newline Suppress Test
set T = $T + 1
echo
echo -e "\n"
echo -N "----> "
echo -N "One"
echo -N "Two"
echo -N "Three"
echo " <----"
echo -e "\n"
read -P "Do you see OneTwoThree between arrows on only ONE line above: " Z
if [ $Z = "y" ]
  set Z = "Y"
fi
if [ $Z = "Y" ]
  if [ -N $log ]
    set P = $P + 1
    echo "echo Newline Suppress Passed" >> ${log}
  fi
else
  if [ -N $log ]
    set F = $F + 1
    echo "echo Newline Suppress Failed" >> ${log}
  fi
fi
echo
echo -e "\necho Tests Done"
if [ -N $log ]
  echo "echo Tests Complete" >> ${log}
fi
set Z =
MAN
TEXT /MAKE/usr/share/tests/echotest
