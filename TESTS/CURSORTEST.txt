#!/bin/sh
#  This script tests various cursor related commands

#  Clear Screen function
function CS {
  ECHO -e \f
}

# Title Box function
# CALL TBOX Str_Title
function TBOX {
  ECHO -e -N "\e[1;20H"
  ECHO -e "\e[7m                                        \e[0m"
  SET LINE = 2
  WHILE [ $LINE -NE 5 ]
    ECHO -e -N "\e[${LINE};20H"
    ECHO -e "\e[7m \e[0m                                      \e[7m \e[0m"
    SET LINE = $LINE + 1
  LOOP
  ECHO -e -N "\e[5;20H"
  ECHO -e "\e[7m                                        \e[0m"
  ECHO -e -N "\e[3;22H $1"
}

# Print Inverse
function INVERSE {
  ECHO -e -N "\e[7m$1\e[0m"
}

# Print at Screen Location function
# CALL PRINTXY Num_Row Num_Column Str_String
function PRINTXY {
  ECHO -e -N "\e[$1;$2H$3"
}

# Print Inverse At X Y
# CALL INVERSEXY Row Col String
function INVERSEXY {
  ECHO -e -N "\e[$1;$2H\e[7m$3\e[0m"
}

# STATUSLINE function
# CALL STATUSLINE Str_Message
function STATUSLINE {
  ECHO -e -N "\e[24;0H\e[7m                                        "
  ECHO -e -N "                                        \e[0m"
  ECHO -e -N "\e[24;2H\e[7m$1\e[0m"
  ECHO -e -N "\e[24;68H\e[7m"
  TIME | CUT -M 10 -N 20
  # SET DTIME = `TIME | CUT -M 10 -N 20`: ECHO -N ${DTIME}
  ECHO -e -N "\e[0m"
}
