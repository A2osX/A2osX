#!/bin/sh
#
#
echo -e "\f\n\n      cat Test"
echo -e "\nThis script will perform tests of the cat command"
echo -e "\nThe first cats a file with line numbers at the left.\n"
read -P "Press Return to start test" Z
if [ -N $log ]
  echo "Begin cat Tests" >> ${log}
else
  set T = 0
  set F = 0
  set P = 0
fi
set T = $T + 1
echo -e "\n"
cat -N cattext
echo -e "\n"
read -P "Did you see 8 numbered lines: " Z
echo
if [ $Z = "y" ]
  set Z = "Y"
fi
if [ $Z = "Y" ]
  if [ -N $log ]
    set P = $P + 1
    echo "cat -N Passed" >> ${log}
  fi
else
  if [ -N $log ]
    set F = $F + 1
    echo "cat -N Failed" >> ${log}
  fi
fi
echo -e "\nThis next test cats the same file but removes"
echo -e "the duplicate empty lines\n"
read -P "Press Return to start test" Z
echo -e "\n"
set T = $T + 1
cat -N -S cattext
echo -e "\n"
read -P "Did you see just 5 lines: " Z
echo
if [ $Z = "y" ]
  set Z = "Y"
fi
if [ $Z = "Y" ]
  if [ -N $log ]
    set P = $P + 1
    echo "cat -S Passed" >> ${log}
  fi
else
  if [ -N $log ]
    set F = $F + 1
    echo "cat -S Failed" >> ${log}
  fi
fi
echo -e "\nThis next test cats a special file and displays"
echo -e "its contents in code.\n"
read -P "Press Return to start test" Z
echo -e "\n"
set T = $T + 1
cat -A catchars
echo -e "\n\n"
echo "The following should have been displayed above:"
echo "[ENQ][ACK][BEL][BS][TAB][VT][FF][SI][DLE][DC1][DC2][DC3]"
echo "[DC4][NAK][SYN][ETB][CAN][EM][SUB][ESC][FS][GS][RS][US]"
echo
read -P "Is that what was displayed: " Z
echo
if [ $Z = "y" ]
  set Z = "Y"
fi
if [ $Z = "Y" ]
  if [ -N $log ]
    set P = $P + 1
    echo "cat -A Passed" >> ${log}
  fi
else
  if [ -N $log ]
    set F = $F + 1
    echo "cat -A Failed" >> ${log}
  fi
fi
echo -e "\nEnd of cat Test"
if [ -N $log ]
  echo "End of cat Test" >> ${log}
fi
set C1 =
set D1 =
set Z =
MAN
TEXT /MAKE/usr/share/tests/cattest
