#include <stdio.h>
#include <stdlib.h>
#include <libtui.h>

#ifndef _TESTLIB_

#define EV_SAVE				192
#define EV_QUIT				193
#define EV_CHECKBOX			194
#define EV_RADIO			195
#define EV_LBOX_CHANGED		196
#define EV_LBOX_PRESSED		197
#define EV_OK				198
#define EV_CANCEL			199

#define _TESTLIB_ 1
#endif

short int LBoxVar;
short int CBoxVar;
short int RadioVar;

int main(int argc, char *argv[])
    {
    tuiCTX pCTX = tuiInit(100);
    tuiOBJ pSCRN = tuiNewScrn(pCTX, F_bTitle+F_bStatus);
    tuiSetProp(pSCRN, P_pTitle, "Screen Title");
    tuiSetProp(pSCRN, P_pStatus, "Status bar");

    char *pBuf = malloc(1024);
    pBuf[0]=0;
    tuiOBJ pTBOX = tuiNewTBox(pSCRN, 1, 1, 44, 20,
	F_bTitle+F_bStatus+F_bHBorder+F_bVBorder+F_bEdit,
	pBuf, 1024);
    tuiSetProp(pTBOX, P_pTitle, "Text Box Title");
    tuiSetProp(pTBOX, P_pStatus, "Text Box Status");

    tuiNewLBox(pSCRN, 50, 1, 20, 9,
		EV_LBOX_CHANGED, EV_LBOX_PRESSED,
		F_bHBorder+F_bVBorder,
        "Item 1\r"
        "Item 2\r"
        "Item 3\r"
        "Item 4\r"
        "Item 5\r"
        "Item 6 very long line\r"
        "Item 7\r"
        "Item 8\r"
        "Item 9\r"
        "Item 10\r"
        "Item 11\r"
        "Item 12\r"
        "Item 13\r"
        "Item 14 last one",
        &LBoxVar);

    char LineBuf[65];
    LineBuf[0]=0;
    tuiNewLabel(pSCRN, 50, 11, "This is a label.");
    tuiNewTLine(pSCRN, 50, 12, 25, &LineBuf, 65);

    tuiNewRadio(pSCRN, 50, 14, 0, 0,
        "\e[91mRed\r"
        "\e[92mGreen\r"
        "\e[96mBlue",
        &RadioVar);

    tuiNewCBox(pSCRN, 50, 18, 0, 0, "Check Me", &CBoxVar);
    tuiNewBut(pSCRN, 50, 20, 19, EV_SAVE, "(^S)ave...");
    tuiNewBut(pSCRN, 65, 20, 17, EV_QUIT, "(^Q)uit");

    tuiDraw(pSCRN);
    tuiActivate(pTBOX);

    short int e;
    do {
        e = tuiExec(pSCRN);
        if (e == EV_SAVE) {
            tuiOBJ pDLGS = tuiNewDlg(pSCRN, 70, 11, "Save As...");
            tuiOBJ pOKBut = tuiNewBut(pDLGS, 50, 7, 13, EV_OK, " OK ");
            tuiOBJ pCanBut = tuiNewBut(pDLGS, 57, 7, 3, EV_CANCEL, "Cancel");
            tuiDraw(pDLGS);
            tuiActivate(pOKBut);
            short int e;
			do {
				e = tuiExec(pDLGS);
				} while (e != EV_CANCEL);
            tuiDestroy(pDLGS);
            tuiDraw(pSCRN);
            }
        } while (e != EV_QUIT);


    tuiDestroy(pSCRN);
    tuiClose(pCTX);
    }
