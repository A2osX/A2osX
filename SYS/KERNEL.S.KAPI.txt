* # GetMemStat
*  Y,A = Ptr to S.MSTAT buffer
* ## RETURN VALUE
*  Buffer filled with memory stats
*\--------------------------------------
				.DUMMY
				.OR K.S.STAT
STAT.AuxBnk		.BS 2					LoMem
				.BS 2                   Free
				.BS 2                   HiMem
				.BS 1                   LastSlot
				.BS 1					BnkID

STAT.U			.BS 4
				.BS 4
				.BS 4

STAT.BT			.BS 2
STAT.BU			.BS 2
STAT.BF			.BS 2

				.ED
*--------------------------------------
K.GetMemStat	>STYA FORPNT

				ldx #11

.1				stz	STAT.U,x
				dex
				bpl .1

				lda #$ff

.2				inc
				sta (pRWReg)
				pha
				jsr RAMSW.GetBnkStat

				jsr STAT.Add2Total

				pla
				cmp A2osX.MaxBnk
				bne .2

*				lda A2osX.ActBnk		Done by LC.GetBM
*				sta (pRWReg)

				ldx #5

.3				stz STAT.BT,x
				dex
				bpl .3

				ldy #$ff

.4				iny

				jsr LC.GetBM

				ldx #8

.5				>INCW STAT.BT
				lsr
				bcs .50

				>INCW STAT.BU
				bra .51

.50				>INCW STAT.BF

.51				dex
				bne .5

				cpy A2osX.MaxBnk
				bne .4

				sta IO.SETWRITEAUX

				ldy #S.MSTAT.MaxBnk
				lda A2osX.MaxBnk
				sta (FORPNT),y

				ldx #11
				ldy #S.MSTAT.UT+11

.6				lda STAT.U,x
				sta (FORPNT),y
				dey
				dex
				bpl .6

				ldx #7
				ldy #S.MSTAT.KL+7

.7				lda MEM.LoMem,x
				sta (FORPNT),y
				dey
				dex
				bpl .7

				ldx #5
				ldy #S.MSTAT.BT+5

.8				lda STAT.BT,x
				sta (FORPNT),y
				dey
				dex
				bpl .8

				sta IO.CLRWRITEAUX

				clc
				rts
*--------------------------------------
STAT.Add2Total	sec
				lda STAT.AuxBnk+4
				sbc STAT.AuxBnk+0
				sta STAT.BT
				lda STAT.AuxBnk+5
				sbc STAT.AuxBnk+1
				sta STAT.BT+1

				sec
				lda STAT.AuxBnk+4
				sbc STAT.AuxBnk+2
				sta STAT.BU
				lda STAT.AuxBnk+5
				sbc STAT.AuxBnk+3
				sta STAT.BU+1

				sec
				lda STAT.BT
				sbc STAT.BU
				sta STAT.BF
				lda STAT.BT+1
				sbc STAT.BU+1
				sta STAT.BF+1

				ldy #0
				ldx #0
				jsr STAT.Add2T

				ldy #2
				ldx #4
				jsr STAT.Add2T

				ldy #4
				ldx #8
*--------------------------------------
STAT.Add2T		clc
				lda STAT.BT,y
				adc STAT.U,x
				sta STAT.U,x

				lda STAT.BT+1,y
				adc STAT.U+1,x
				sta STAT.U+1,x

				bcc .8

				inc STAT.U+2,x

*				bne .8

*				inc STAT.U+3,x

.8				rts

K.SPCall		ldy #3					ROM entry point

				>STYA RAMSW.2SP+4

				ldy #2					SPCall #

				sta RAMSW.2SP+6

				ldy #0					Param list
				jsr RAMSW.GetStkYW
				>STYA RAMSW.2SP+7

				sta IO.SETWRITEAUX

				sta IO.SET80STORE		Make sure Screen holes in MAIN

				jsr RAMSW.2SP

				sta IO.CLR80STORE

				sta IO.CLRWRITEAUX

				rts
