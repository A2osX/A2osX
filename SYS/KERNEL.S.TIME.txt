* # Time
* Get System Time in Buffer
* ## C
* `#include <time.h>`
* `time_t time(time_t *tloc);`
* ## ASM
* `>SL`
* `>LDYA tloc`
* `>LIBC time`
* ## RETURN VALUE
*  time_t
*\--------------------------------------
K.Time			>STYA FORPNT			tloc

				>MLICALL MLI.GETTIME

				ldy #3

.1				lda MLI.DATE,y
				sta TIMEX.PT,y
				dey
				bpl .1

				ldx #X_PT2TM
				jsr JMPX

				ldx #X_TM2CT
				jsr JMPX

				ldy #3

				sta IO.SETWRITEAUX

				lda FORPNT+1
				beq .3

.2				lda TIMEX.CT,y
				sta (FORPNT),y
				dey
				bpl .2

				ldy #3

.3				lda TIMEX.CT,y
				sta (pStack),y
				dey
				bpl .3

				sta IO.CLRWRITEAUX

				clc
				rts
*/--------------------------------------
* # MKTime
*  Convert S.TM to CTime
* ## C
* `#include <prodos.h>`
* `time_t mktime(struct tm *timeptr);`
* ## ASM
* `>SL`
* `>LDYA timeptr`
* `>LIBC MKTime`
* ## RETURN VALUE
*\--------------------------------------
K.MKTime		>STYA TXTPTR			timeptr

				ldy #S.TM-1

.1				jsr RAMSW.xTXTPTRgY
				sta TIMEX.TM,y
				dey
				bpl .1

				ldx #X_TM2CT
				jsr JMPX

				sta IO.SETWRITEAUX

				ldy #3

.2				lda TIMEX.CT,y
				sta (pStack),y
				dey
				bpl .2

				sta IO.CLRWRITEAUX

				clc
				rts
*/--------------------------------------
* # GMTime
*  Convert CTime To S.TM
* ## C
* `#include <time.h>`
* `struct tm *gmtime(const time_t *timer);`
* ## ASM
* `>LDYA timeptr`
* `>LIBC CTime2Time`
* ## RETURN VALUE
*  Y,A = pointer to allocatd tm struct
*\--------------------------------------
K.GMTime		>STYA TXTPTR

				ldy #3

.1				jsr RAMSW.xTXTPTRgY
				sta ARG32,y				ARG = ctime
				dey
				bpl .1

				ldx #X_CT2TM
				jsr JMPX

				>LDYAI S.TM
				jsr K.Malloc
				bcs .9

				>STYA FORPNT

				ldy #S.TM-1

				sta IO.SETWRITEAUX

.2				lda TIMEX.TM,y
				sta (FORPNT),y
				dey
				bpl .2

				sta IO.CLRWRITEAUX

				>LDYA FORPNT

*				clc

.9				rts
*/--------------------------------------
* # StrFTime
* ## C
* Convert S.TM struct to CSTR
* `#include <time.h>`
* `void strftime (char* str, const char* format, const struct tm* timeptr);`
* ## ASM
* `>SS`
* `>PUSHW str`
* `>PUSHW format`
* + %a : Abbreviated weekday name : Thu
* + %A : Full weekday name : Thursday
* + %b : Abbreviated month name : Aug
* + %B : Full month name : August
* + %d : Day of the month, zero-padded (01-31)
* + %H : Hour in 24h format (00-23)
* + %I : Hour in 12h format (01-12)
* + %m : Month as a decimal number (01-12)
* + %M : Minute (00-59)
* + %p : AM or PM designation
* + %S : Second (00-61)
* + %w : Weekday as a decimal number with Sunday as 0 (0-6)
* + %y : Year, last two digits (00-99)
* + %Y : Year four digits (2025)
* `>PUSHW timeptr`
* `>LIBC strftime`
* `>SR`
* ## RETURN VALUE
*   none. always succeed.
*\--------------------------------------
K.StrFTime		ldy #0					S.TM
				jsr RAMSW.StkY2TXTPTR

				ldy #S.TM-1

.10				jsr RAMSW.xTXTPTRgY
				sta TIMEX.TM,y
				dey
				bpl .10

				ldy #2					format
				jsr RAMSW.StkY2TXTPTR

				stz TIMEX.S

.1				jsr RAMSW.xTXTPTRgn
				beq .3

				cmp #'%'
				beq .2

				ldx TIMEX.S
				sta TIMEX.S+1,x
				inc TIMEX.S
				bra .1

.2				jsr RAMSW.xTXTPTRgn
				beq .3

				tay

				ldx #X_ST2S
				jsr JMPX
				bra .1

.3				ldx TIMEX.S
				sta TIMEX.S+1,x

				ldy #4					str
				jsr RAMSW.StkY2FORPNT

				ldy #$ff

				sta IO.SETWRITEAUX

.4				iny
				lda TIMEX.S+1,y
				sta (FORPNT),y
				bne .4

				sta IO.CLRWRITEAUX

				clc
				rts
