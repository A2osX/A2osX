* ## MD5
* Return MD5 Hash for input String
* # C
* `void md5 (const char* str, char* digest);`
* # ASM
* `>SS`
* `>PUSHW str`
* `>PUSHW digest`
* `>LIBC md5`
* `>SR`
* ## RETURN VALUE
* CC
*\--------------------------------------
K.MD5			ldy #2					str
				jsr RAMSW.StkY2TXTPTR

				ldy #$ff

.1				iny
				jsr RAMSW.xTXTPTRgY
				bne .1

				sty ZPDataLen
				stz ZPDataLen+1

				ldx #X_MD5
				jsr JMPX

				ldy #0
				jsr RAMSW.StkY2FORPNT	digest

				sta IO.SETWRITEAUX

				ldy #$ff

.2				iny
				lda FOUTBuf,y
				sta (FORPNT),y
				bne .2

				sta IO.CLRWRITEAUX

				clc
				rts
*/--------------------------------------
* ## MD5Init
* Initialize a MD5 computation
* # C
* `void *md5init();`
* # ASM
* `>LIBC MD5Init`
* ## RETURN VALUE
* Y,A = pMD5
*\--------------------------------------
K.MD5Init		ldx #X_MD5I
				jmp JMPX

* ## MD5Update
* Add Data to MD5 computation
* # C
* `void md5update (void *md5, char *data, int len);`
* # ASM
* `>SS`
* `>PUSHW md5`
* `>PUSHW data`
* `>PUSHW len`
* `>LIBC MD5Update`
* `>SR`
* ## RETURN VALUE
*\--------------------------------------
K.MD5Update		ldy #4					md5
				jsr RAMSW.GetStkYW
				>STYA ZPPtr3

				ldy #2					data
				jsr RAMSW.StkY2TXTPTR

				ldy #0					len
				jsr RAMSW.GetStkYW
*				jsr RAMSW.StkY2FORPNT

*				>LDYA FORPNT
				>STYA ZPDataLen

				ldx #X_MD5U
				jmp JMPX
*/--------------------------------------
* ## MD5Finalize
* # C
* `void md5finalize (void *md5, char* digest);`
* # ASM
* `>SS`
* `>PUSHW md5`
* `>PUSHW digest`
* `>LIBC MD5Finalize`
* `>SR`
* ## RETURN VALUE
*\--------------------------------------
K.MD5Finalize	ldy #2					md5
				jsr RAMSW.GetStkYW
				>STYA ZPPtr3

				ldy #0					digest
				jsr RAMSW.StkY2FORPNT

				ldx #X_MD5F
				jmp JMPX
