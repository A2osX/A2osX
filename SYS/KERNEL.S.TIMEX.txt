SECSDAY			.EQ 86400				60*60*24
CENT0			.EQ 19
YEAR0			.EQ 70
DAY0			.EQ 4					day 0 was a thursday
*--------------------------------------
TIMEX.E			lda #E.OOB
				sec
				rts
*--------------------------------------
TIMEX.PT2TM		lda TIMEX.PT+3
				and #%11100000
				bne TIMEX.E

				lda TIMEX.PT+2
				and #%11000000
				bne TIMEX.E

				lda TIMEX.PT+1			Get Year
				lsr						C is high bit of month
				sta TIMEX.TM+S.TM.YEAR

				sta ARG32+3				for computing Century/WDAY later

				lda TIMEX.PT+0			Get Month/day

				tax						save Day
				ror						C is high bit of month
				lsr
				lsr
				lsr
				lsr
				beq TIMEX.E

				cmp #13
				bcs TIMEX.E				range not in 1..12

				sta TIMEX.TM+S.TM.MON

				txa						get back day
				and #$1F
				beq TIMEX.E				range not in 1..31

				sta TIMEX.TM+S.TM.MDAY

				dec						adjust range 0.30 for MOD 7
				sta ARG32S				For later...

				lda ARG32+3				get back year
				cmp #100
				bcs TIMEX.E

				cmp #69					if before 70 CC,if > 70, CS
				lda #0
				rol						get back C in bit 0
				eor #1					toggle C
				adc #19					set date before 1970 -> 20xx
				sta TIMEX.TM+S.TM.CENT
				sta ARG32+2				for computing WDAY later

				lda TIMEX.PT+3			Get Hour
				cmp #24
				bcs TIMEX.E

				sta TIMEX.TM+S.TM.HOUR

				lda TIMEX.PT+2			Get Min
				cmp #60
				bcs TIMEX.E

				sta TIMEX.TM+S.TM.MIN

				stz TIMEX.TM+S.TM.SEC	set seconds (ProDOS does not provide it)
*--------------------------------------
* 1/1/1970 was Thursday...if not leap, add one, if leap add 2
*--------------------------------------
TIMEX.TMWDAY	lda #DAY0-1				Thursday : 4 (-1 for mod 7)
				pha

				lda #CENT0				Starts at 1970
				sta ARG32
				lda #YEAR0
				sta ARG32+1

.1				ldy ARG32
				lda ARG32+1

				cpy ARG32+2
				bne .2

				cmp ARG32+3
				beq .4

.2				jsr TIMEX.IsLeap		CS = Leap

				pla
				adc #1					365 mod 7
				cmp #7
				bcc .3

				sbc #7					MOD 7

.3				pha
				inc ARG32+1
				lda ARG32+1
				cmp #100
				bne .1

				stz ARG32+1
				inc ARG32
				bra .1
*--------------------------------------
.4				ldx TIMEX.TM+S.TM.MON

				dex						don't include actual month
				beq .71

.5				clc						assume NO leap

				lda TIMEX.MDAY-1,x		get day count in this month
				bne	.6					february ?

				ldy ARG32+2
				lda ARG32+3
				jsr TIMEX.IsLeap		CS = Leap

.6				pla
				adc TIMEX.MDAY-1,x

				cmp #7
				bcc .7

				sbc #7

.7				pha
				dex
				bne .5
*--------------------------------------
.71				pla
				clc

				adc ARG32S			get day in month (0..30)

.8				cmp #7					MOD 7
				bcc .80

				sbc #7
				bra .8

.80				inc						adjust range 1..7
				sta TIMEX.TM+S.TM.WDAY

*				clc

				rts
*--------------------------------------
TIMEX.CT2TM		lda #SECSDAY			ACC = SECSDAY
				sta ACC32
				lda /SECSDAY
				sta ACC32+1
				lda ^SECSDAY
				sta ACC32+2
				stz ACC32+3
				jsr M32.DIVMOD  		ARG32 = Days, TMP32 = remaining secs

				lda ARG32				WDAY computation : (ARG32 + DAY0) mod 7
				clc
				adc #DAY0
				pha
				lda ARG32+1				65535 days = 179 years
				adc /DAY0
				eor #$ff
				tax
				pla

.2				tay
				sec
				sbc #7
				bcs .2

				inx
				bne .2

				sty TIMEX.TM+S.TM.WDAY

				ldy #CENT0
				sty ZPPtr3
				lda #YEAR0
				sta ZPPtr3+1
TIMEX.CT2TM.Y
.1				ldy ZPPtr3
				lda ZPPtr3+1
				jsr TIMEX.IsLeap 		if Leap year CS
				rol						Toggle Carry
				eor #1
				lsr
				lda ARG32
				sbc #365
				pha
				lda ARG32+1
				sbc /365
				bcc .2

				sta ARG32+1
				pla
				sta ARG32
				inc ZPPtr3+1
				lda ZPPtr3+1
				cmp #100
				bne .1

				stz ZPPtr3+1
				inc ZPPtr3
				bra .1

.2				pla

				lda ZPPtr3
				sta TIMEX.TM+S.TM.CENT
				lda ZPPtr3+1
				sta TIMEX.TM+S.TM.YEAR

TIMEX.CT2TM.MD	ldx #1					X = January

.1				clc
				txa
				eor #2

				bne .2

				>LDYA ZPPtr3			Current Year
				jsr TIMEX.IsLeap

.2				lda TIMEX.MDAY-1,x
				adc #28					A = days of X

				sta .3+1

				lda ARG32				ARG32/ARG32+1 = 0 .. 364 or 365
				sec

.3				sbc #$ff				SELF MODIFIED
				pha
				lda ARG32+1
				sbc #0
				bcc .4

				sta ARG32+1
				pla
				sta ARG32
				inx
				bra .1

.4				pla
				stx TIMEX.TM+S.TM.MON

				lda ARG32
				inc
				sta TIMEX.TM+S.TM.MDAY

TIMEX.CT2TM.HMS	ldx #3

.1				lda TMP32,x
				sta ARG32,x
				dex
				bpl .1

				lda #60
				jsr M32.A2ACC32
				jsr M32.DIVMOD
				lda TMP32
				sta TIMEX.TM+S.TM.SEC

				lda #60
				jsr M32.A2ACC32
				jsr M32.DIVMOD
				lda TMP32
				sta TIMEX.TM+S.TM.MIN

				lda ARG32
				sta TIMEX.TM+S.TM.HOUR

				clc
				rts

TIMEX.TM2CT		lda TIMEX.TM+S.TM.CENT
				cmp #19
				bne .1

				stz TIMEX.CT
				stz TIMEX.CT+1
				stz TIMEX.CT+2
				stz TIMEX.CT+3
				bra .2

.1				cmp #20
				beq .55

				lda #E.OOB
				sec
				rts


* 2000/01/01 00:00:00 = 946684800

.55				lda #946684800
				sta TIMEX.CT
				lda /946684800
				sta TIMEX.CT+1
				lda ^946684800
				sta TIMEX.CT+2
				lda ^946684800/256
				sta TIMEX.CT+3

.2				stz ZPPtr3

.3				lda ZPPtr3
				cmp TIMEX.TM+S.TM.YEAR
				beq .5

				inc ZPPtr3

				ldy TIMEX.TM+S.TM.CENT
				jsr TIMEX.IsLeap
				bcc .4

				ldx #8						1 day
				jsr TIMEX.CTAddX

.4				ldx #0
				jsr TIMEX.CTAddX			1 year
				bra .3

.5				stz ZPPtr3

.6				inc ZPPtr3

				ldy ZPPtr3
				cpy TIMEX.TM+S.TM.MON
				beq .10

				lda TIMEX.MDAY-1,y			get day count in this month
				bne .7

				lda TIMEX.TM+S.TM.YEAR
				ldy TIMEX.TM+S.TM.CENT
				jsr TIMEX.IsLeap
				bcc .9

				lda #1

.7				sta ZPPtr3+1

.8				ldx #8						1 day
				jsr TIMEX.CTAddX
				dec	ZPPtr3+1
				bne .8

.9				ldx #4						28 days
				jsr TIMEX.CTAddX
				bra .6

.10				dec TIMEX.TM+S.TM.MDAY
				beq .11

				ldx #8						1 day
				jsr TIMEX.CTAddX
				bra .10

.11				dec TIMEX.TM+S.TM.HOUR
				bmi .12

				ldx #12						1 hour
				jsr TIMEX.CTAddX
				bra .11

.12				dec TIMEX.TM+S.TM.MIN
				bmi .13

				lda #60
				jsr TIMEX.CTAddA
				bra .12

.13				lda TIMEX.TM+S.TM.SEC
*--------------------------------------
TIMEX.CTAddA	clc
				adc TIMEX.CT
				sta TIMEX.CT
				bcc .8

				inc TIMEX.CT+1
				bne .8

				inc TIMEX.CT+2
				bne .8

				inc TIMEX.CT+3

.8				rts
*--------------------------------------
TIMEX.CTAddX	ldy #0

				clc

.1				lda .2,x
				adc TIMEX.CT,y
				sta TIMEX.CT,y
				inx
				iny
				tya
				eor #$4
				bne .1

				rts

.2				.DA	>31536000			1 year
				.DA >2419200			28 days
				.DA >86400				1 day
				.DA >3600				1 hour
*--------------------------------------
* In :
*  Y = Century
*  A = Year (0..99)
*    if (year mod 400 = 0)
*      or
*    if  not (year mod 100 = 0) and (year mod 4 = 0)
* Out :
*  CS = Leap
*  CC = Not Leap
*--------------------------------------
TIMEX.IsLeap	cmp #0					Year = 00 ?
				bne .1					no

				tya						year = 00, get century in A

				and #$3					Century mod 4 = 0 ?
				beq .9					leap year (2000)

.8				clc						not leap (1900)
				rts

.1				and #$3					mod 4 = 0 ?
				bne .8

.9				sec						Leap
				rts

TIMEX.ST2S		tya

				ldx #TIMEX.ST2S.Cnt-1

.3				cmp TIMEX.ST2S.Tbl,x
				beq .4

				dex
				bpl .3

				rts

.4				cpx #7
				bcc .5

				jmp TIMEX.ST2S.AtX

.5				txa
				asl
				tax

.6				jmp (.7,x)

.7				.DA TIMEX.ST2S.A
				.DA TIMEX.ST2S.AA
				.DA TIMEX.ST2S.B
				.DA TIMEX.ST2S.BB
				.DA TIMEX.ST2S.II
				.DA TIMEX.ST2S.P
				.DA TIMEX.ST2S.YY
*--------------------------------------
TIMEX.ST2S.Tbl	.AS "aAbBIpYydHSwmM"
TIMEX.ST2S.Cnt	.EQ *-TIMEX.ST2S.Tbl
TIMEX.ST2S.OfsX	.DA #S.TM.YEAR,#S.TM.MDAY,#S.TM.HOUR,#S.TM.SEC,#S.TM.WDAY,#S.TM.MON,#S.TM.MIN
*--------------------------------------
TIMEX.ST2S.A	sec						Short day of week, 3 chars...
				.HS	90					BCC
TIMEX.ST2S.AA	clc						full DoW
				lda #TIMEX.DAY
				ldx /TIMEX.DAY
				ldy TIMEX.TM+S.TM.WDAY
				bra TIMEX.ST2S.STR

TIMEX.ST2S.B	sec						Short Month, 3 chars....
				.HS	90					BCC
TIMEX.ST2S.BB	clc						full Month....
				lda #TIMEX.MON
				ldx /TIMEX.MON
				ldy TIMEX.TM+S.TM.MON

TIMEX.ST2S.STR	sta ZPPtr4
				stx ZPPtr4+1

				ldx #15
				bcc .10

				ldx #3

.10				tya
				beq .2					Illegal value, print ???

.1				lda (ZPPtr4)
				sec
				adc ZPPtr4
				sta ZPPtr4
				bcc .11

				inc ZPPtr4+1

.11				dey
				bne .1
.2
*				ldy #0					Y is already 0

.3				iny
				lda (ZPPtr4),y
				jsr TIMEX.2S
				tya
				cmp (ZPPtr4)
				beq .8

				dex
				bne .3

.8				rts
*--------------------------------------
TIMEX.ST2S.II	lda TIMEX.TM+S.TM.HOUR
				cmp #12
				bcc TIMEX.ST2S.addDecA

				sbc #12
				bra TIMEX.ST2S.addDecA

TIMEX.ST2S.P	lda TIMEX.TM+S.TM.HOUR
				cmp #12
				bcc .1

				lda #'p'
				.HS 2C					bit abs
.1				lda #'a'
				jsr TIMEX.2S
				lda #'m'
				bra TIMEX.2S

TIMEX.ST2S.YY	lda TIMEX.TM+S.TM.CENT
				jsr TIMEX.ST2S.addDecA

				ldx #7					"y"

TIMEX.ST2S.AtX	ldy TIMEX.ST2S.OfsX-7,x
				lda TIMEX.TM,y
*--------------------------------------
TIMEX.ST2S.addDecA
				ldx #$30

.1				cmp #10
				bcc .8

				sbc #10
				inx
				bra .1

.8				ora #$30

				pha
				txa

				jsr TIMEX.2S

				pla

				rts
