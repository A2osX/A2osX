CODE.Init		>STZ.G CC.bInitCode

				>STA.G CC.ScopeID
				iny
				sta (pData),y

				ldy #SCOPE-1
				lda #0

.1				sta (ScopeStk),y		reset initial Scope ID, LocalPtr, FrameSize
				dey
				bpl .1

				>LDYAI PCC.FH.CS
				>STYA ZPCCCode

				stz ZPCCConst
				stz ZPCCConst+1

				>LDYAI 20				char *ARGV[10]
				>STYA ZPCCData

				clc

				rts
*--------------------------------------
CODE.Quit		>LDYA.G CC.pOutFile
				cmp #0
				beq .8

				>LIBC FClose

.8				clc
				rts
*--------------------------------------
CODE.AddLocal
* for this to work 3 passes required
*				cmp #0
*				beq .8

				eor #$FF
				inc

				jsr CODE.LDAI
				bcs .9

				>LDYA L.PCC.ADDLOCAL
				jmp CODE.EmitPCC

.8				clc
.9				rts
*--------------------------------------
CODE.RemLocal
* for this to work 3 passes required
*				cmp #0
*				beq .8

				pha

				lda #$AA				TAX
				jsr CODE.EmitByte

				pla
				jsr CODE.LDAI
				bcs .9

				>LDYA L.PCC.REMLOCAL
				jmp CODE.EmitPCC

.8				clc
.9				rts
*--------------------------------------
CODE.SetRetValue
				jsr CODE.LDXI
				bcs .9

				>LDYA L.PCC.SetRetValue
				jmp CODE.EmitPCC

.9				rts

CODE.GetIAddr1

CODE.GetUAddr1	lda #$48				PHA
				jsr CODE.EmitByte
				bcs .9

				ldy #SYM.Addr
				lda (ZP.pLSymBuf),y
				jsr CODE.LDAI
				bcs .9

				>LDYA L.PCC.GetUAddrL1
				jsr CODE.EmitPCC
				bcs .9

				ldy #SYM.Addr+1
				lda (ZP.pLSymBuf),y
				jsr CODE.LDAI
				bcs .9

				>LDYA L.PCC.GetUAddrH1
				jmp CODE.EmitPCC

.9				rts

CODE.PUSHAXI	pha
				txa
				jsr CODE.PUSHI
				bcs CODE.LDAXI.RTS

				pla
*--------------------------------------
CODE.PUSHI		jsr CODE.LDAI
				bcs CODE.LDAXI.RTS

				bra CODE.PUSHA
*--------------------------------------
CODE.LDYAI		pha

				lda #$A0				LDY #imm
				jsr CODE.EmitByte
				bcs CODE.LDAXI.RTS

				tya
				jsr CODE.EmitByte
				bcs CODE.LDAXI.RTS

				lda #$A9				LDA #imm
				jsr CODE.EmitByte
				bcs CODE.LDAXI.RTS

				pla

				jmp CODE.EmitByte
*--------------------------------------
CODE.LDAXI		jsr CODE.LDAI
				bcc CODE.LDXI

CODE.LDAXI.RTS	rts
*--------------------------------------
CODE.LDYI		lda #$A0				LDY #imm
				jsr CODE.EmitByte
				bcs .9

				tya
				jmp CODE.EmitByte

.9				rts
*--------------------------------------
CODE.LDXI		lda #$A2				LDX #imm
				jsr CODE.EmitByte
				bcs .9

				txa
				jmp CODE.EmitByte

.9				rts
*--------------------------------------
CODE.LDAI		pha
				lda #$A9				LDA #imm
				jsr CODE.EmitByte
				bcs .9

				pla
				jmp CODE.EmitByte

.9				plx
				rts
*--------------------------------------
CODE.PUSHYA		jsr CODE.PUSHA
				lda #$98				TYA
				jsr CODE.EmitByte
*--------------------------------------
CODE.PUSHA		lda #$C6				DEC zp
				jsr CODE.EmitByte
				lda #pStack
				jsr CODE.EmitByte

				lda #$92				STA (zp)
				jsr CODE.EmitByte
				lda #pStack
				jmp CODE.EmitByte
*--------------------------------------
CODE.PULLYA		jsr CODE.PULLA
				lda #$48				PHA
				jsr CODE.EmitByte
				jsr CODE.PULLA
				lda #$7A				PLY
				bra CODE.EmitByte
*--------------------------------------
CODE.PULLA		lda #$B2				LDA (zp)
				jsr CODE.EmitByte
				lda #pStack
				jsr CODE.EmitByte
*--------------------------------------
CODE.INCPSTACK	lda #$E6				INC zp
				jsr CODE.EmitByte
				lda #pStack
				bra CODE.EmitByte
*--------------------------------------

				bra CODE.TOABSYX

.9				rts

				bcs .9

				tya
				dec
				dec
				dec

				adc #pLibJs
				tay

				lda (pData),y
				pha
				iny

				tay
				plx

				lda #$20				JSR

				ldy #0

.1				iny
				lda (ZP.pLookup),y

				bcs .9

				tya

				bne .1

.9				rts

CODE.EmitByte	.DO _DBG_CSOUT=1

				phy
				phx
				pha
				tay

				>SS
				>PUSHW L.MSG.DEBUG.CSOUT
				>PUSHW ZPCCCode
				tya
				>PUSHA
				>PUSHBI 3
				>LIBC PrintF
				>SR

				pla
				plx
				ply

				.FIN

				clc						if Pass 1...
				bit bPass2
				bpl .1

				jsr FIO.WriteA
				bcs .99

.1				>INCW ZPCCCode

.99				rts						CC/CS from FPutC
*--------------------------------------
CODE.EmitData	.DO _DBG_IDOUT=1

				phy
				phx
				pha
				tay

				>SS
				>PUSHW L.MSG.DEBUG.IDOUT
				lda PCC.FH+PCC.FH.CS.END
				clc
				adc ZPCCConst
				tax
				lda PCC.FH+PCC.FH.CS.END+1
				adc ZPCCConst+1
				>PUSHA
				txa
				>PUSHA
				tya
				>PUSHA
				>PUSHBI 3
				>LIBC PrintF
				>SR

				pla
				plx
				ply

				.FIN

				clc
				bit bPass2
				bpl .1

				jsr FIO.WriteA
				bcs .99

.1				>INCW ZPCCConst

.99				rts						CC/CS from FPutC

				sta PCC.FH+PCC.FH.DS.SIZE+1
