LINK.GetLIBs	stz ArgIndex

				>LDYA PCC.FH+PCC.FH.ID.END
				>STYA ZPPtr1

.1				lda ArgIndex
				>CMP.G LibCount
				beq .8

				asl
				clc
				adc #pLibJs
				tay
				lda ZPPtr1
				sta (pData),y
				iny
				lda ZPPtr1+1
				sta (pData),y

				lda ArgIndex
				asl
				clc
				adc #pLibNs
				tay
				lda (pData),y
				sta ZPPtr2
				iny
				lda (pData),y
				sta ZPPtr2+1

				inc ArgIndex

				ldy #$ff

.3				iny
				lda (ZPPtr2),y
				bne .3

				tya
				sec						+ \0
				adc ZPPtr1
				sta ZPPtr1
				bcc .1

				inc ZPPtr1+1
				bra .1

.8				clc

.99				rts
*--------------------------------------
LINK			ldy #0
				lda #2

.1				>SS
				>STYA ZPPtr1			SymID
				>PUSHW.G CC.pSyms
				>PUSHW ZPPtr1			SymID
				>PUSHW pData
				>KAPI SListGetByID
				>SR
				bcs .8

				>STYA ZPPtr2			NextID

				>SS
				>PUSHW.G CC.pSyms
				>PUSHW ZPPtr1			SymID
				>PUSHW ZP.pLSymBuf
				>PUSHWI 65535			All
				>PUSHWZ					from Start
				>KAPI SListGetData
				>SR
				bcs .99

				>LDA.G bVerbose
				bpl .2

				jsr LINK.Dump

.2				ldy #SYM.SC
				lda (ZP.pLSymBuf),y
				bne .7					SYM.SC.STATIC

				ldy #SYM.Addr+1
				lda (ZP.pLSymBuf),y
				dey
				ora (ZP.pLSymBuf),y
				bne .7

				jsr LINK.Lookup
				bcs .9

				jsr LINK.Update
				bcs .99

.7				>LDYA ZPPtr2
				jmp .1

.8				clc

				rts

.9				>LDA.G bVerbose
				bmi .90

				jsr LINK.Dump

.90				lda #E.FUNDEF
				sec
.99				rts
*--------------------------------------
* Input : pData = Symbol
*--------------------------------------
LINK.Lookup		>LDYA L.CC.LIBC
				jsr LINK.LookupYA
				ldx #1
				bcc .99

				>LDA.G LibCount
*				sec
				beq .99

				lda #0
				sta ArgIndex
				clc

.1				asl
				adc #pLibOs
				tay
				lda (pData),y
				pha
				iny
				lda (pData),y
				ply

				jsr LINK.LookupYA
				bcs .2

				ldx ArgIndex
				inx
				inx
				inx
				rts

.2				inc ArgIndex
				lda ArgIndex
				>CMP.G LibCount
				bcc .1

.99				rts
*--------------------------------------
LINK.LookupYA	>STYA ZP.pLookup

				ldy #$ff

.1				iny
				lda (pData),y
				bne .1

				tya
				tax						Symbol Len

.2				lda (ZP.pLookup)
				beq .9					Ending 0, not found....

				txa						Get Back Len
				cmp (ZP.pLookup)		Same Len ?
				bne .4

				tay

.3				lda (ZP.pLookup),y

				dey
				cmp (pData),y
				bne .4

				tya
				bne .3

				lda (ZP.pLookup)
				tay
				iny
				lda (ZP.pLookup),y			get fID

				clc
				rts

.4				lda (ZP.pLookup)
				inc							Skip LEN\Symbol\ID

				sec
				adc ZP.pLookup
				sta ZP.pLookup
				bcc .2

				inc ZP.pLookup+1
				bra .2

.9				sec
				rts
*--------------------------------------
* A=ID X=SYS|LIB
*--------------------------------------
LINK.Update		ldy #SYM.Addr
				sta (ZP.pLSymBuf),y

				iny
				txa
				sta (ZP.pLSymBuf),y

				>SS
				>PUSHW.G CC.pSyms
				>PUSHW ZPPtr1			SymID
				>PUSHW ZP.pLSymBuf
				>PUSHWI SYM
				>KAPI SListSetData
				>SR
				rts
*--------------------------------------
LINK.Dump		ldy #$ff

.1				iny
				lda (pData),y
				bne .1

				iny
				lda (pData),y
				sta ZPPtr3
				iny
				lda (pData),y
				sta ZPPtr3+1

				>SS
				>PUSHW L.MSG.LINKING2

				>PUSHW ZPPtr1			ID
				>PUSHW ZPPtr3			Scope

				>PUSHW pData			Name

				>PUSHB (ZP.pLSymBuf)	SYM.T

				ldy #1
				>PUSHB (ZP.pLSymBuf),y	SYM.Q

				iny
				>PUSHB (ZP.pLSymBuf),y	SYM.SC

				ldy #SYM.Cnt
				>PUSHB (ZP.pLSymBuf),y

				ldy #SYM.SizeOf+1
				>PUSHB (ZP.pLSymBuf),y
				dey
				>PUSHB (ZP.pLSymBuf),y

				ldy #SYM.Addr+1
				>PUSHB (ZP.pLSymBuf),y
				dey
				>PUSHB (ZP.pLSymBuf),y

				>PUSHBI 14
				>LIBC PrintF
				>SR
				rts

				sta PCC.FH+PCC.FH.MAIN+1
				dey
				lda (ZP.pLSymBuf),y
				sta PCC.FH+PCC.FH.MAIN

*				clc

				rts

.9				>LDYA L.MSG.NOMAIN
				>LIBC PutS
				lda #E.CSYN
				sec
.99				rts
