NEW
  AUTO 3,1
				.LIST OFF
				.OP	65C02
				.OR	$2000
				.TF bin/md5
*--------------------------------------
* passowrd = 5f4dcc3b5aa765d61d8327deb882cf99
*--------------------------------------
				.INB inc/macros.i
				.INB inc/a2osx.i
				.INB inc/a2osx.api.i
				.INB inc/mli.e.i
*--------------------------------------
FILEBUF.SIZE	.EQ 4096
*--------------------------------------
				.DUMMY
				.OR ZPBIN
ZS.START
ZPPtr1			.BS 2

ArgIndex		.BS 1
Arg				.BS 1
bText			.BS 1

pMD5Ctx			.BS 2

pFile			.BS 1

ZPDataBuf		.BS 2
ZPDataLen		.BS 2

ZS.END			.ED
*--------------------------------------
*			File Header (16 Bytes)
*--------------------------------------
CS.START		cld
				jmp (.1,x)
				.DA #$61				6502,Level 1 (65c02)
				.DA #2					BIN Layout Version 2
				.DA #0					#S.PS.SIG
				.DA #0
				.DA CS.END
				.DA ID.END
				.DA DS.END
				.DA #64					Stack Size
				.DA #ZS.END-ZS.START	Zero Page Size
*--------------------------------------
* Relocation Table
*--------------------------------------
.1				.DA CS.INIT
				.DA CS.RUN
				.DA CS.SIG
				.DA	CS.QUIT
*--------------------------------------
L.MSG.USAGE		.DA MSG.USAGE
				.DA 0
*--------------------------------------
CS.INIT			clc
				rts
*--------------------------------------
CS.RUN			ldy #S.PS.ARGC
				jsr A2osX.GetPSy
				beq .99

.1				inc ArgIndex
				lda ArgIndex
				>KAPI ArgV
				bcs .8

				>STYA ZPPtr1
				lda (ZPPtr1)
				cmp #'-'
				bne .4

				ldy #1
				lda (ZPPtr1),y
				beq .99

				ldy #OptionVars-OptionList-1

.2				cmp OptionList,y
				beq .3

				dey
				bpl .2

.99				>LDYA L.MSG.USAGE
				>LIBC PutS
				lda #E.SYN
				sec
				rts

.3				ldx OptionVars,y
				sec
				ror 0,x
				bra .1

.4				lda Arg
				bne .99

				lda ArgIndex
				sta Arg
				bra .1

.8				lda Arg
				beq .99

				lda bText
				beq CS.RUN.FILE
*--------------------------------------
CS.RUN.TEXT		lda Arg
				>KAPI ArgV

				>SS
				>PUSHYA
				>PUSHEA.G MD5Buf
				>LIBC MD5
				>SR
				jmp CS.RUN.PRINT
*--------------------------------------
CS.RUN.FILE		>LDYAI FILEBUF.SIZE
				>LIBC Malloc
				bcs .9

				>STYA ZPDataBuf

				lda Arg
				>KAPI ArgV

				>SS
				>PUSHYA
				>PUSHBI	O.RDONLY
				>PUSHBI 0				ftype
				>PUSHWZ					Aux type
				>LIBC FOpen
				>SR
				bcs .9

				>STYA pFile

				>LIBC MD5Init
				bcs .9

				>STYA pMD5Ctx

.1				>SLEEP

				>SS
				>PUSHW pFile
				>PUSHW ZPDataBuf
				>PUSHWI FILEBUF.SIZE	Bytes To Read
				>LIBC FRead
				>SR
				bcc .2

				cmp #MLI.E.EOF
				beq .8

.9				rts

.2				>STYA ZPDataLen

				>SLEEP

				>SS
				>PUSHW pMD5Ctx
				>PUSHW ZPDataBuf
				>PUSHW ZPDataLen
				>LIBC MD5Update
				>SR
				bra .1

.8				>SS
				>PUSHW pMD5Ctx
				>PUSHEA.G MD5Buf
				>LIBC MD5Finalize
				>SR

CS.RUN.PRINT	>LEA.G MD5Buf
				>LIBC PutS
				lda #0
*				sec
*				rts
*--------------------------------------
CS.SIG			rts
*--------------------------------------
CS.QUIT			>LDYA ZPDataBuf
				beq .2

				>LIBC Free

.2				>LDYA pFile
				beq .3

				>LIBC FClose

.3				clc
				rts
*--------------------------------------
CS.END			.EQ *
*--------------------------------------
OptionList		.AS "Dd"
OptionVars		.DA #bText,#bText
*--------------------------------------
MSG.USAGE		.CZ "Usage : MD5 [ -d input text | input file ]\r\n"
*--------------------------------------
ID.END			.EQ *
*--------------------------------------
				.DUMMY
				.OR 0
MD5Buf			.BS 33
DS.END			.ED
*--------------------------------------
MAN
SAVE usr/src/bin/md5.s
ASM
