SCOPE.New		lda ScopePtr
				tay
				clc
				adc #SCOPE
				bcs .9

				sta ScopePtr

				iny
				iny						#SCOPE.LocalPtr
				lda (ScopeStk),y
				pha

				>LDA.G CC.ScopeID
*				clc
				adc #1
				sta (pData),y
				tax
				iny
				lda (pData),y
				adc #0

				ldy ScopePtr
				iny
				sta (ScopeStk),y		#SCOPE.ID Hi
				txa
				dey
				sta (ScopeStk),y		#SCOPE.ID Lo

				iny
				iny
				pla
				sta (ScopeStk),y		#SCOPE.LocalPtr

				iny
				lda #0
				sta (ScopeStk),y		#SCOPE.ParamSize
				iny
				sta (ScopeStk),y		#SCOPE.FrameSize
				iny
				sta (ScopeStk),y		#SCOPE.State

				.DO _DBG_SCOPE=1
			>SS
			>PUSHW L.MSG.DEBUG.SCIN
			lda ScopePtr
			>PUSHA

			jsr SCOPE.Push

			tay
			iny
			iny
			>PUSHB (ScopeStk),y
			iny
			>PUSHB (ScopeStk),y
			iny
			>PUSHB (ScopeStk),y
			iny
			>PUSHB (ScopeStk),y

			>PUSHBI 7
			>LIBC PrintF
			>SR
				.FIN

				clc
				rts

.9				lda #E.OOH
				sec
				rts
*--------------------------------------
SCOPE.Get		lda ScopePtr
				tay
				clc
				adc #SCOPE
				sta ScopePtr

				iny
				iny						#SCOPE.LocalPtr
				lda (ScopeStk),y
				pha

				>LDA.G CC.ScopeID
				tax
				iny
				lda (pData),y

				ldy ScopePtr
				iny
				sta (ScopeStk),y		#SCOPE.ID Hi
				txa
				dey
				sta (ScopeStk),y		#SCOPE.ID Lo

				iny
				iny

				pla

				sta (ScopeStk),y		#SCOPE.LocalPtr

				iny
				lda #0
				sta (ScopeStk),y		#SCOPE.ParamSize
				iny
				sta (ScopeStk),y		#SCOPE.FrameSize
				iny
				sta (ScopeStk),y		#SCOPE.State

				.DO _DBG_SCOPE=1
			>SS
			>PUSHW L.MSG.DEBUG.SCIN
			lda ScopePtr
			>PUSHA

			jsr SCOPE.Push

			tay
			iny
			iny
			>PUSHB (ScopeStk),y
			iny
			>PUSHB (ScopeStk),y
			iny
			>PUSHB (ScopeStk),y
			iny
			>PUSHB (ScopeStk),y

			>PUSHBI 7
			>LIBC PrintF
			>SR
				.FIN

				rts
*--------------------------------------
SCOPE.AddLocal	lda #'F'
				jsr SYM.LookupLabelA
				bcs .9

				txa

				jmp CODE.AddLocal

.9				rts
*--------------------------------------
SCOPE.RemLocal	lda #'F'				define FRAME size
				jsr SYM.NewLabelA
				bcs .9

				lda ScopePtr
*				clc
				adc #SCOPE.FrameSize
				tay
				lda (ScopeStk),y

				jmp CODE.RemLocal

.9				rts
*--------------------------------------
SCOPE.Close		.DO _DBG_SCOPE=1

			>SS
			>PUSHW L.MSG.DEBUG.SCOUT
			lda ScopePtr
			>PUSHA

			jsr SCOPE.Push

			tay
			iny
			iny
			>PUSHB (ScopeStk),y
			iny
			>PUSHB (ScopeStk),y
			iny
			>PUSHB (ScopeStk),y
			iny
			>PUSHB (ScopeStk),y

			>PUSHBI 7
			>LIBC PrintF
			>SR
				.FIN

				lda ScopePtr
				beq *

				sec
				sbc #SCOPE
				sta ScopePtr

				.DO _DBG_SCOPE=1
			>SS
			>PUSHW L.MSG.DEBUG.SC

			lda ScopePtr
			>PUSHA

			jsr SCOPE.Push

			tay
			iny
			iny
			>PUSHB (ScopeStk),y
			iny
			>PUSHB (ScopeStk),y
			iny
			>PUSHB (ScopeStk),y
			iny
			>PUSHB (ScopeStk),y

			>PUSHBI 7
			>LIBC PrintF
			>SR

				.FIN

				clc

.99				rts

SCOPE.Push		lda ScopePtr

SCOPE.PushA		clc
				adc #SCOPE.ID+1
				tay
				lda (ScopeStk),y
				>PUSHA
				dey						#SCOPE.ID
				lda (ScopeStk),y
				>PUSHA
				rts
