

NEW
  AUTO 3,1
				.LIST OFF
				.OP	65C02
				.OR	$2000
				.TF bin/zcutil
*--------------------------------------
NSC.IOBASE		.EQ $C300
*--------------------------------------
				.INB inc/macros.i
				.INB inc/io.i
				.INB inc/io.zc.i
				.INB inc/a2osx.i
				.INB inc/a2osx.api.i
				.INB inc/mli.e.i
*--------------------------------------
				.DUMMY
				.OR ZPBIN
ZS.START
ZPPtr1			.BS 2
ZS.END			.ED
*--------------------------------------
*			File Header (16 Bytes)
*--------------------------------------
CS.START		cld
				jmp (.1,x)
				.DA #$61				6502,Level 1 (65c02)
				.DA #2					BIN Layout Version 2
				.DA #0					S.PS.F.EVENT
				.DA #0
				.DA CS.END
				.DA ID.END
				.DA DS.END
				.DA #64					Stack Size
				.DA #ZS.END-ZS.START	Zero Page Size
*--------------------------------------
* Relocation Table
*--------------------------------------
.1				.DA CS.INIT
				.DA CS.RUN
				.DA CS.SIG
				.DA	CS.QUIT
*--------------------------------------
L.MSG.DETECT.KO	.DA MSG.DETECT.KO
L.MSG.DETECT.OK	.DA MSG.DETECT.OK

L.MSG.USAGE		.DA MSG.USAGE
L.MSG.CRLF		.DA MSG.CRLF
				.DA 0
*--------------------------------------
CS.INIT
*--------------------------------------
CS.SIG
*--------------------------------------
CS.QUIT			clc
				rts
*--------------------------------------
CS.RUN			jsr ZC.Detect
				bcc .1

				>LDYA L.MSG.DETECT.KO
				>LIBC PutS
				lda #MLI.E.NODEV
				sec
				rts

.1				lda #1
				>KAPI ArgV
				bcc CS.RUN.SET

CS.RUN.READ		jsr ZC.Read

				jsr ZC.Dump

				lda #0
				sec
				rts
*--------------------------------------
CS.RUN.SET		>STYA ZPPtr1

				>LDYA L.MSG.USAGE
				>LIBC PutS

.9				lda #E.SYN
				sec
				rts
*--------------------------------------
ZC.Dump			>SS
				>PUSHW L.MSG.DETECT.OK

				>PUSHB ZC.STATUS
				>PUSHB ZC.SLOW
				>PUSHB ZC.SPEED
				>PUSHB ZC.LCPDL

				>PUSHBI 4
				>LIBC PrintF
				>SR

				clc
				rts
*--------------------------------------
ZC.Detect		php
				sei

				lda #IO.ZC.ACCESS.UNLOCK
				sta IO.ZC.ACCESS
				sta IO.ZC.ACCESS
				sta IO.ZC.ACCESS
				sta IO.ZC.ACCESS

				lda IO.ZC.STATUS
				sta IO.ZC.CTRL

				ldx #0

.1				lda .2,x
				inx
				cpx #.8-.2
				bcc .1

.2				lda IO.ZC.SLOW
				eor #$ff
				sta IO.ZC.SLOW
				cmp IO.ZC.SLOW
				bne .9

				eor #$ff
				sta IO.ZC.SLOW
				cmp IO.ZC.SLOW
				bne .9

				lda #IO.ZC.ACCESS.LOCK
				sta IO.ZC.ACCESS

.8				plp
				clc
				rts

.9				plp
				sec
				rts
*--------------------------------------
ZC.Read			php
				sei

				lda #IO.ZC.ACCESS.UNLOCK
				sta IO.ZC.ACCESS
				sta IO.ZC.ACCESS
				sta IO.ZC.ACCESS
				sta IO.ZC.ACCESS

				lda IO.ZC.STATUS
				sta IO.ZC.CTRL

				ldx #0

.1				lda .2,x
				inx
				cpx #.8-.2
				bcc .1

.2				lda IO.ZC.STATUS
				sta ZC.STATUS

				lda IO.ZC.SLOW
				sta ZC.SLOW

				lda IO.ZC.SPEED
				sta ZC.SPEED

				lda IO.ZC.LCPDL
				sta ZC.LCPDL

				lda #IO.ZC.ACCESS.LOCK
				sta IO.ZC.ACCESS

.8				plp
				rts
*--------------------------------------
ZC.Write		php
				sei


.8				plp
				rts
*--------------------------------------
CS.END			.EQ *
*--------------------------------------
MSG.DETECT.KO	.CZ "ZipChip Not Detected.\r\n"
MSG.DETECT.OK	.CS "ZipChip Detected :\r\n"
				.CS " STATUS : %b\r\n"
				.CS " SLOW   : %b\r\n"
				.CS " SPEED  : %b\r\n"
				.CZ " LCPDL  : %b\r\n"

MSG.USAGE		.CS "zcutil -S xxx -L y...."
MSG.CRLF		.CZ "\r\n"

MSG.SPEEDS		.CZ "4.0000"
				.CZ "2.6667"
				.CZ "3.0000"
				.CZ "3.2000"
				.CZ "3.3333"
				.CZ "2.0000"
				.CZ "1.3333"
				.CZ "1.5000"
				.CZ "1,6000"
				.CZ "1.6667"
				.CZ "0.6667"
				.CZ "0.7500"
				.CZ "0.8000"
				.CZ "0.8333"
				.CZ "1.3333"
				.CZ "0.8889"
				.CZ "1.0000"
				.CZ "1.0667"
				.CZ "1.1111"

*--------------------------------------
ZC.STATUS		.BS 1
ZC.SLOW			.BS 1
ZC.SPEED		.BS 1
ZC.LCPDL		.BS 1
*--------------------------------------
ID.END			.HS 00					no lib
*--------------------------------------
                .DUMMY
                .OR 0
DS.END          .ED
*--------------------------------------
MAN
SAVE usr/src/bin/zcutil.s
ASM
