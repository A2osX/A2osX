NEW
  AUTO 3,1
				.LIST OFF
				.OP	65C02
				.OR	$2000
				.TF bin/lsfd
*--------------------------------------
				.INB inc/macros.i
				.INB inc/a2osx.i
				.INB inc/a2osx.api.i
				.INB inc/io.i
*--------------------------------------
				.DUMMY
				.OR ZPBIN
ZS.START
hFD				.BS 2
pFD				.BS 2
pName			.BS 2
Tmp				.BS 1
Err				.BS 1

ZS.END			.ED
*--------------------------------------
*			File Header (16 Bytes)
*--------------------------------------
CS.START		cld
				jmp (.1,x)
				.DA #$61				6502,Level 1 (65c02)
				.DA #2					BIN Layout Version 2
				.DA #0					S.PS.SIG
				.DA #0
				.DA CS.END
				.DA ID.END
				.DA DS.END
				.DA #64					Stack Size
				.DA #ZS.END-ZS.START	Zero Page Size
*--------------------------------------
* Relocation Table
*--------------------------------------
.1				.DA CS.INIT
				.DA CS.RUN
				.DA CS.SIG
				.DA	CS.QUIT
*--------------------------------------
L.MSG.Header	.DA MSG.Header
L.MSG.FD		.DA MSG.FD
L.MSG.DEV		.DA MSG.DEV
L.MSG.DEV.ERR	.DA MSG.DEV.ERR
L.MSG.DEV.Long	.DA MSG.DEV.Long
L.MSG.DEV.NA	.DA MSG.DEV.NA
*--------------------------------------
J.FD			.DA DumpFD.REG
				.DA DumpFD.DIR
				.DA DumpFD.CDEV
				.DA DumpFD.CDEV
				.DA DumpFD.LNK
				.DA DumpFD.DSOCK
				.DA DumpFD.SSOCK
				.DA DumpFD.PIPE
*--------------------------------------
L.FD.T			.DA FD.T.REG
				.DA FD.T.DIR
				.DA FD.T.CDEV
				.DA FD.T.BDEV
				.DA FD.T.LNK
				.DA FD.T.DSOCK
				.DA FD.T.SSOCK
				.DA FD.T.PIPE
				.DA 0
*--------------------------------------
CS.INIT
CS.SIG
CS.QUIT			clc
				rts
*--------------------------------------
CS.RUN			>LDYA L.MSG.Header
				>LIBC PutS

				lda #0

.1				sta hFD
				>KAPI GetFDByID
				bcs .7

				>STYA pFD

				stz pName
				stz pName+1

				lda hFD
				>KAPI GetNameByID
				bcs .4

				>STYA pName

.4				jsr DumpFD

				>LDYA pFD
				>LIBC Free

				>LDYA pName
				beq .7

				>LIBC Free

.7				lda hFD

				inc
				inc
				cmp #K.FD.MAX*2
				bcc .1

				lda #0

*				sec

.9				rts
*--------------------------------------
DumpFD			>SS
				>PUSHW L.MSG.FD

				>PUSHB hFD

				ldy #S.FD.T
				lda (pFD),y
				tax
				>PUSHW L.FD.T,x

				ldy #S.FD.O
				lda (pFD),y

				ldy #7

.1				lsr
				pha
				lda #'-'
				bcc .2

				lda MSG.OFLAGS,y

.2				sta (pData),y			DATABUF
				pla
				dey
				bpl .1

				ldy #8
				lda #0
				sta (pData),y			DATABUF

				>PUSHW pData

				ldy #S.FD.OCNT+1
				lda (pFD),y
				>PUSHA
				dey
				lda (pFD),y
				>PUSHA

				>PUSHBI 7

				>LIBC PrintF
				>SR
				bcs .9

				jsr DumpFD1
				bcs .9

				>LDYA pName
				>LIBC PutS

.9				rts
*--------------------------------------
DumpFD1			lda (pFD)				#S.FD.T

				tax
				jmp (J.FD,x)

DumpFD.REG		>SS
				>PUSHW L.MSG.DEV.Long

				ldy #S.FD.REG.FPOS+3

.1				lda (pFD),y
				>PUSHA
				dey
				cpy #S.FD.REG.FPOS-1

				>PUSHBI 4
				>LIBC PrintF
				>SR

				rts
*--------------------------------------
DumpFD.DIR
DumpFD.CDEV
DumpFD.BDEV
DumpFD.LNK
DumpFD.DSOCK
DumpFD.SSOCK
DumpFD.PIPE

				>SS
				>PUSHW L.MSG.DEV.NA
				>PUSHBI 0
				>LIBC PrintF
				>SR

				rts
*--------------------------------------
PrintErr		sta Err

				>SS
				>PUSHB Err
				>PUSHW pData			DATABUF
				>KAPI GetErrMsg
				>SR

				>SS
				>PUSHW L.MSG.DEV.ERR
				>PUSHB Err
				>PUSHW pData			DATABUF
				>PUSHBI 3
				>LIBC PrintF
				>SR
				rts
*--------------------------------------
CS.END			.EQ *
*--------------------------------------
MSG.Header		.CZ "FD Type oFlags   oCnt  size/pos   Name"
MSG.FD			.CZ "%h %4s %s %5D "
MSG.DEV			.CZ "%13s "
MSG.DEV.ERR		.CZ "[%h] %s\r\n"

MSG.DEV.Long	.CZ "%10u \r\n"
MSG.DEV.NA		.CZ "n/a        "
*--------------------------------------
MSG.OFLAGS		.AS "ct--atwr"
*--------------------------------------
FD.T.REG		.AZ "REG"
FD.T.DIR		.AZ "DIR"
FD.T.CDEV		.AZ "CDEV"
FD.T.BDEV		.AZ "BDEV"
FD.T.LNK		.AZ "LNK"
FD.T.DSOCK		.AZ "DSCK"
FD.T.SSOCK		.AZ "SSCK"
FD.T.PIPE		.AZ "PIPE"
*--------------------------------------
ID.END			.HS 00					no lib
*--------------------------------------
				.DUMMY
				.OR 0

DATABUF			.BS 64
DS.END			.ED
*--------------------------------------
MAN
SAVE usr/src/bin/lsfd.s
ASM
