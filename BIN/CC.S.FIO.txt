FIO.LoadLIB		>STYA ZPPtr1

				>LDA.G LibCount
				cmp #LIBS.MAX
				bne .1

				lda #E.OOH
				sec
.9				rts

.1				asl
				clc
				adc #pLibNs
				tay

				lda ZPPtr1
				sta (pData),y
				iny
				lda ZPPtr1+1
				sta (pData),y

				jsr FIO.MakeLibPath
				bcs .99

				>SS
				>PUSHW L.MSG.READING
				>PUSHW pData
				>PUSHBI 2
				>LIBC PrintF
				>SR
				bcs .99

				>SS
				>PUSHW pData
				>PUSHBI	O.RDONLY
				>PUSHBI S.FI.T.BIN
				>PUSHWZ					Aux type
				>LIBC LoadFile
				>SR
				bcs .99

				pha
				phy

				>LDA.G LibCount
				asl
				clc
				adc #pLibOs
				tay
				pla
				sta (pData),y
				iny
				pla
				sta (pData),y

				>INC.G LibCount
				clc

.99				rts
*--------------------------------------
FIO.MakeLibPath	>SS
				>LDYA L.ENV.LIB
				>LIBC GetEnv
				>SR
				bcs .9

				>SS
				pha
				>PUSHW pData			>PUSHEA.G PathBuf
				pla
				>PUSHA
				tya
				>PUSHA
				>LIBC StrCpy
				>SR

				>SS
				>PUSHW pData
				>PUSHW ZPPtr1
				>LIBC StrCat
				>SR

				>SS
				>PUSHW pData
				>PUSHW L.LIB.O
				>LIBC StrCat
				>SR

.9				rts
*--------------------------------------
FIO.FOpen1st	>LDYA.G CC.pInFileName

FIO.FOpen		>STYA ZPPtr1

				>SS
				>PUSHW L.MSG.READING
				>PUSHW ZPPtr1
				>PUSHBI 2
				>LIBC PrintF
				>SR

				>SS
				>PUSHW ZPPtr1
				>PUSHBI	O.RDONLY
				>PUSHBI S.FI.T.TXT
				>PUSHWZ					Aux type
				>LIBC FOpen
				>SR
				bcs .9

				pha
				phy

				>LDA.G CC.pInFiles
				inc
				inc
				sta (pData),y
				tay
				pla
				sta (pData),y
				iny
				pla
				sta (pData),y

				>LDA.G CC.LineCnts
				inc
				inc
				sta (pData),y
				tay
				lda #0
				sta (pData),y
				iny
				sta (pData),y

.9				rts
*--------------------------------------
FIO.FGetS		>LDYA.G CC.pLineBuf
				>STYA ZP.pLineBuf

				lda #0
				sta (ZP.pLineBuf)		set to Empty line

				>LDA.G CC.LineCnts
				tay
				lda (pData),y
				inc
				sta (pData),y
				bne .1

				iny
				lda (pData),y
				inc
				sta (pData),y

.1				>SS
				>PUSHW ZP.pLineBuf
				>PUSHWI 255
				>LDA.G CC.pInFiles
				tay

				iny
				lda (pData),y
				>PUSHA
				dey
				lda (pData),y
				>PUSHA

				>LIBC FGetS
				>SR
				bcs .9

				>LDA.G bVerbose
				bpl .8

				jsr PrintVerboseMsg

.8

.9				rts
*--------------------------------------
FIO.FClose		>LDA.G CC.pInFiles
				cmp #CC.pInFiles
				beq .8					Z

				tay
				lda (pData),y
				pha
				iny
				lda (pData),y
				ply

				>LIBC FClose

				>LDA.G CC.pInFiles
				dec
				dec
				sta (pData),y
				cmp #CC.pInFiles
				beq .8					Z

				>LDA.G CC.LineCnts
				dec
				dec
				sta (pData),y			NZ

.8				rts
*--------------------------------------
FIO.CreateBIN	>SS
				>PUSHW L.MSG.GENERATING
				>PUSHW.G CC.pOutFileName
				>PUSHBI 2
				>LIBC PrintF
				>SR
				bcs .99

				>SS
				>PUSHW.G CC.pOutFileName
				>PUSHBI	O.WRONLY+O.CREATE
				>PUSHBI S.FI.T.BIN
				>PUSHWI	$2000			Aux type
				>LIBC FOpen
				>SR
				bcs .99

				>STYA.G CC.pOutFile

				>SS
				>PUSHW.G CC.pOutFile
				>PUSHWZ
				>PUSHWI PCC.FH.L
				>PUSHWI SEEK.SET
				>LIBC FSeek
				>SR

.99				rts
*--------------------------------------
FIO.CloseBIN	jsr FIO.WriteLIBNames
				bcs .99

				jsr FIO.WriteHDR
				bcs .99


.99				rts
*--------------------------------------
FIO.WriteLIBNames
				jsr FIO.IDSelect
				bcs .99

				stz ArgIndex

.1				lda ArgIndex
				>CMP.G LibCount
				beq .8

				asl
				clc
				adc #pLibNs
				tay
				lda (pData),y
				sta ZPPtr1
				iny
				lda (pData),y
				sta ZPPtr1+1

				inc ArgIndex

				ldy #$ff

.3				iny
				lda (ZPPtr1),y
				beq .4

				jsr FIO.WriteA
				bcc .3

				rts

.4				jsr FIO.WriteA
				bcc .1
				
.99				rts

.8				lda #0
				jmp FIO.WriteA
*--------------------------------------
FIO.WriteHDR	>SS
				>PUSHW.G CC.pOutFile
				>PUSHWZ
				>PUSHWZ
				>PUSHWI SEEK.SET
				>LIBC FSeek
				>SR
				bcs .99

				>SS
				>PUSHW.G CC.pOutFile
				>PUSHW L.PCC.FH
				>PUSHWI PCC.FH.L
				>LIBC FWrite
				>SR

.99				rts
*--------------------------------------
FIO.CSSelect	>SS
				>PUSHW.G CC.pOutFile
				>PUSHWZ
				lda ZPCCCode+1
				sec
				sbc #$20
				>PUSHA
				lda ZPCCCode
				>PUSHA
				>PUSHWI SEEK.SET
				>LIBC FSeek
				>SR

				rts
*--------------------------------------
FIO.IDSelect	>SS
				>PUSHW.G CC.pOutFile
				>PUSHWZ

				lda PCC.FH+PCC.FH.CS.END
				clc
				adc ZPCCConst
				tay

				lda PCC.FH+PCC.FH.CS.END+1
				adc ZPCCConst+1
				sec
				sbc #$20

				>PUSHYA
				>PUSHWI SEEK.SET
				>LIBC FSeek
				>SR

				rts

*--------------------------------------

				phy

				>SS
				dec pStack				WORD
				>PUSHA
				>PUSHW.G CC.pOutFile
				>LIBC FPutC
				>SR

				ply
				plx
				rts
