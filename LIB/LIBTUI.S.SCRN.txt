* # tuiNewScrn
* Initialize screen & create a new context
* ## C
* `tuiOBJ tuiNewScrn(tuiOBJ, Flags);`
* ## ASM
* `>SS`
* `>PUSHW tuiOBJ`
* `>PUSHW Flags`
* `>LIBCALL LIBTUI,tuiNewScrn`
* `>SR`
* ## RETURN VALUE
* CC = success, Y,A = Event
*\--------------------------------------
SCRN.New		lda #S.OBJ.T.SCRN
				jsr OBJ.Create
				bcs .99

				>GETSTKW
				ldy #S.OBJ.F
				sta (ZPpObj),y

				>GETSTKW 2
				>STAX ZPpCtx
				ldy #S.OBJ.pParent
				sta (ZPpObj),y
				txa
				iny
				sta (ZPpObj),y

				lda ZPpObj
				sta (ZPpCtx)			S.CTX.pSCRN

				lda ZPpObj+1
				ldy #S.CTX.pSCRN+1
				sta (ZPpCtx),y

				ldy #S.CTX.W
				lda (ZPpCtx),y
				ldy #S.OBJ.W
				sta (ZPpObj),y

				ldy #S.CTX.H
				lda (ZPpCtx),y
				ldy #S.OBJ.H
				sta (ZPpObj),y

				jsr OBJ.SetX2Y2



				>LDYA ZPpObj
*				clc

.99				rts
