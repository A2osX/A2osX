NEW
  PREFIX
AUTO 4,1
#!/bin/sh
#
#  This script makes a proper FULL32 32mb disk using the files
#  found in a particular BUILD stored in /make/builds
#
. makefuncs

set TVOL = "FULL32"
set BVOL = "MKBLD"
set TBLANK = "BLANK32"
set TDRIVE = "s7d2"
set VTEST = "vtest.tmp"

#echo "Star before: $*"
#call getArgs
#echo "ALL_YES=${ALL_YES}, RCS=${RCS}, B=${B}"
#echo "Star after: $*"
#pause

# if they specified build number on command line
if [ $# = 1 ]
    set B = $1
fi

CALL CS
CALL TBOX "   ${TVOL} 32mb Media Creator"
CALL PRINTXY 8 0 "Looking for blank media ${TBLANK} on /dev/${TDRIVE}"
ECHO -e "\n"

# make sure our target volume isn't named the same as current ROOT
if [ ${ROOT} = "/${TVOL}/" ]
    call INVERSE "WARNING:"
    echo " Boot volume has same name as target volume; Scripts and/or A2osX may become"
    echo -e "         unusable after install completes!\n"
fi


# make sure BLANKxxx is mounted and tweak TDRIVE if needed
call findSlotDrive ${TBLANK} ${TDRIVE}
if [ -n ${A} ]
    echo "WARNING: Did not find /${TBLANK} volume. Assuming ${TDRIVE} as target drive."
    echo "         ${A} on target drive will be overwritten!"
else
    echo "Found /${TBLANK} at ${SLDR}, will use as target drive."
    set TDRIVE = ${SLDR}
fi

READ -P "Do you want to proceed (Y/N)?  " P
SWITCH $P
  CASE Y
  CASE y
    ECHO
    BREAK
  DEFAULT
    ECHO -e "\n\nExiting.\n"
    EXIT
END

if [ -z $B ]
    READ -P "WHICH BUILD NO:  " B
    IF [ -Z $B ]
      ECHO "You did not enter any response."
      ECHO
      ECHO "Exiting"
      EXIT
    FI
fi

SET SRC = "/MAKE/BUILDS/BUILD${B}"
IF [ -D ${SRC} ]
  ECHO -e "\n\nFound Valid BUILD ${B}\n"
ELSE
  ECHO -e "\n\nValid BUILD ${B} not found."
  ECHO
  ECHO "Exiting"
  EXIT
FI
ECHO "Is this a (R)elease, (C)andidate or (S)table Build"
READ -P "ENTER R C or S:  " RCS
IF [ -Z $RCS ]
  ECHO "You did not enter any response."
  ECHO
  ECHO "Exiting"
  EXIT
FI
SWITCH $RCS
  CASE r
  CASE R
    SET IM = "Release Build No. ${B}"
    BREAK
  CASE c
  CASE C
    SET IM = "Release Candidate Build No. ${B}"
    BREAK
  CASE s
  CASE S
    SET IM = "Stable Release Build No. ${B}"
    BREAK
  DEFAULT
    ECHO "You did not enter a valid response."
    ECHO  "You must enter only one of the characters R C or S!"
    ECHO
    ECHO "Exiting"
    EXIT
END
ECHO "Formatting Destination..."
format -B 65535 s7d2 ${BVOL}
ECHO "Making Directory Structure..."
md /${BVOL}/bin
md /${BVOL}/drv
md /${BVOL}/etc
md /${BVOL}/inc
md /${BVOL}/include
md /${BVOL}/include/sys
md /${BVOL}/lib
md /${BVOL}/root
md /${BVOL}/root/acostest
md /${BVOL}/root/ftest
md /${BVOL}/root/shtest
md /${BVOL}/root/cshtest
md /${BVOL}/sbin
md /${BVOL}/sys
md /${BVOL}/sys/km
md /${BVOL}/sys/pm
md /${BVOL}/tmp
md /${BVOL}/usr
md /${BVOL}/usr/share
md /${BVOL}/usr/share/adm
md /${BVOL}/usr/share/extras
md /${BVOL}/var
md /${BVOL}/var/bbs
md /${BVOL}/var/log
md /${BVOL}/var/www
ECHO "Copying Files..."
FOR FILE IN (fullfiles)
  if [ -f ${SRC}/${FILE} ]
#    echo "- ${SRC}/${FILE}"
    cp -Q ${SRC}/${FILE} /${BVOL}/${FILE}
  else
    echo "NOT FOUND: ${SRC}/${FILE}"
  fi
NEXT
cp -Q -Y -R ${SRC}/srv/www/* /${BVOL}/var/www
cp -Q -Y -R ${SRC}/var/bbs/* /${BVOL}/var/bbs
cp -Q -Y ${SRC}/PRODOS.FX /${BVOL}/ProDOS
cp -Q -Y ${SRC}/PRODOS.FX /${BVOL}/usr/share/extras
cp -Q -Y ${SRC}/PRODOS.FX.PAKED /${BVOL}/usr/share/extras
cp -Q -Y ${SRC}/PRODOS.203TC /${BVOL}/usr/share/extras
cp -Q -Y ${SRC}/BASIC.SYSTEM /${BVOL}/bin/basic
cp -Q -Y /MAKE/assets/NS.CLOCK.SYSTEM /${BVOL}/usr/share/extras
cp -Q -Y -R /MAKE/usr/share/help /${BVOL}/usr/share
cp -Q -Y -R /MAKE/usr/share/man /${BVOL}/usr/share
cp -Q -Y initboot /${BVOL}/etc/init
ECHO "Welcome to A2osX!" > /${BVOL}/etc/issue
ECHO >> /${BVOL}/etc/issue
ECHO $IM >> /${BVOL}/etc/issue
echo "${B}" > /${BVOL}/etc/version
cp -Q tcpip.conf /${BVOL}/etc/tcpip.conf
cp -Q profile /${BVOL}/root/.profile
cp -q helloworld.s /${BVOL}/root/helloworld.s
cp -q -r -y /MAKE/usr/share/adm/adm /${BVOL}/bin/adm
cp -q -r -y /MAKE/usr/share/adm/help /${BVOL}/bin/help
cp -q -r -y /MAKE/usr/share/adm/update /${BVOL}/usr/share/adm/update

# we built to BVOL, now rename to proper TVOL
ren /${BVOL} ${TVOL}
echo "${TVOL} 32mb Disk Created!"
MAN
TEXT /MAKE/usr/share/make/makefull32
