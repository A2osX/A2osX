  PREFIX
AUTO 4,1
#!/bin/sh
#
#  This script makes a proper FULL800 disk using the files
#  found in a particular build stored in /make/builds
#
. makefuncs

set TBLANK = "BLANK800"
set TVOL = "FULL800"
set TDRIVE = "s5d1"
set VTEST = "vtest.tmp"

# if they specified build number on command line
if [ $# = 1 ]
    set B = $1
fi

CALL CS
CALL TBOX "   ${TVOL} 800K Media Creator"
CALL PRINTXY 8 0 "Looking for blank media ${TBLANK} on /dev/${TDRIVE}"
ECHO -e "\n"

# make sure BLANKxxx is mounted and tweak TDRIVE if needed
call findSlotDrive ${TBLANK} ${TDRIVE}
if [ -n ${A} ]
    echo "WARNING: Did not find /${TBLANK} volume. Assuming ${TDRIVE} as target drive."
    echo "         ${A} on target drive will be overwritten!"
else
    echo "Found /${TBLANK} at ${SLDR}, will use as target drive."
    set TDRIVE = ${SLDR}
fi


READ -P "Do you want to proceed (Y/N)?  " P
SWITCH $P
  CASE Y
  CASE y
    ECHO
    BREAK
  DEFAULT
    ECHO -e "\n\nExiting.\n"
    EXIT
END

if [ -z $B ]
    READ -P "WHICH BUILD NO:  " B
    IF [ -Z $B ]
      ECHO -e "You did not enter any response.\n\nExiting"
      EXIT
    FI
fi

SET SRC = "/MAKE/BUILDS/BUILD${B}"
IF [ -D ${SRC} ]
  ECHO -e "\n\nFound Valid BUILD ${B}\n"
ELSE
  ECHO -e "\n\nValid BUILD ${B} not found.\nExiting\n"
  EXIT
FI
ECHO "Is this a (R)elease, (C)andidate or (S)table Build"
READ -P "ENTER R C or S:  " RCS
IF [ -Z $RCS ]
  ECHO -e "You did not enter any response.\nExiting\n"
  EXIT
FI
SWITCH $RCS
  CASE r
  CASE R
    SET IM = "Release Build No. ${B}"
    BREAK
  CASE c
  CASE C
    SET IM = "Release Candidate Build No. ${B}"
    BREAK
  CASE s
  CASE S
    SET IM = "Stable Release Build No. ${B}"
    BREAK
  DEFAULT
    ECHO "You did not enter a valid response."
    ECHO  "You must enter only one of the characters R C or S!"
    ECHO
    ECHO "Exiting"
    EXIT
END
ECHO "Which Prodos:"
ECHO "    1) ProDOS 2.03tc"
ECHO "    2) ProDOS FX"
ECHO "    3) ProDOS FX Paked"
READ -P "Enter 1-3:  " PDOS
IF [ -Z $PDOS ]
  ECHO "You did not enter any response."
  ECHO
  ECHO "Exiting"
  EXIT
FI
SWITCH $PDOS
  CASE 1
    SET PDOSFILE = "${SRC}/PRODOS"
    BREAK
  CASE 2
    SET PDOSFILE = "${SRC}/PRODOS.FX"
    BREAK
  CASE 3
    SET PDOSFILE = "${SRC}/PRODOS.FX.PAKED"
    BREAK
  DEFAULT
    ECHO "You did not enter a valid response."
    ECHO  "You must enter only one of the characters R C or S!"
    ECHO
    ECHO "Exiting"
    EXIT
END
ECHO "Formatting Destination..."
format -B 1600 ${TDRIVE} ${TVOL}
ECHO "Making Directory Structure..."
md /${TVOL}/bin
md /${TVOL}/drv
md /${TVOL}/etc
md /${TVOL}/inc
md /${TVOL}/lib
md /${TVOL}/root
md /${TVOL}/root/acostest
md /${TVOL}/root/ftest
md /${TVOL}/root/shtest
md /${TVOL}/root/cshtest
md /${TVOL}/sbin
md /${TVOL}/sys
md /${TVOL}/sys/pm
md /${TVOL}/tmp
md /${TVOL}/usr
md /${TVOL}/usr/share
md /${TVOL}/usr/share/adm
md /${TVOL}/usr/share/extras
md /${TVOL}/var
md /${TVOL}/var/bbs
md /${TVOL}/var/log
md /${TVOL}/var/www
ECHO "Copying Files..."
set -e
FOR FILE IN (fullfiles)
  if [ -f ${SRC}/${FILE} ]
#    echo "- ${SRC}/${FILE}"
    cp -Q ${SRC}/${FILE} /${TVOL}/${FILE}
  else
    echo "NOT FOUND: ${SRC}/${FILE}"
  fi
NEXT
cp -Q -Y $PDOSFILE /${TVOL}/ProDOS
cp -Q -Y initboot /${TVOL}/etc/init
ECHO "Welcome to A2osX!" > /${TVOL}/etc/issue
ECHO >> /${TVOL}/etc/issue
ECHO $IM >> /${TVOL}/etc/issue
echo "${B}" > /${TVOL}/etc/version
cp -Q tcpip.conf /${TVOL}/etc/tcpip.conf
cp -Q profile /${TVOL}/root/.profile
IF [ $PDOS -eq 1 ]
  ECHO Renaming VOL
  REN /${TVOL} ${TVOL}
FI
cp -q helloworld.s /${TVOL}/root/helloworld.s
cp -q ${SRC}/BASIC.SYSTEM /${TVOL}/bin/basic
#cp -q /MAKE/SCASM3.SYSTEM /${TVOL}/bin/SC3
cp -q -r -y /MAKE/usr/share/adm/adm /${TVOL}/bin/adm
cp -q -r -y /MAKE/usr/share/adm/help /${TVOL}/bin/help
cp -q -r -y /MAKE/usr/share/adm/update /${TVOL}/usr/share/adm/update
cp -q -r -y /MAKE/usr/share/help /${TVOL}/usr/share
cp -q -r -y /MAKE/usr/share/man /${TVOL}/usr/share

ECHO "${TVOL} Disk Created!"
MAN
TEXT /MAKE/usr/share/make/makefull
