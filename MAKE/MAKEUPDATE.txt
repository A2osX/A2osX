NEW
  PREFIX
AUTO 4,1
#!/bin/sh
#
#  This script makes an UPDATE.PAK file using the files
#  found in a particular BUILD stored in /make/builds
#
. makefuncs

set TVOL = "MAKE/update"

# if they specified build number on command line
if [ $# = 1 ]
    set B = $1
fi

CALL CS
CALL TBOX "      Update PAK Creator"

echo
if [ -z $B ]
    READ -P "WHICH BUILD NO:  " B
    IF [ -Z $B ]
      ECHO -e "You did not enter any response.\n\nExiting"
      EXIT
    FI
fi

SET SRC = "/MAKE/BUILDS/BUILD${B}"
IF [ -D ${SRC} ]
  ECHO -e "\n\nFound Valid BUILD ${B}\n"
ELSE
  ECHO -e "\n\nValid BUILD ${B} not found."
  ECHO
  ECHO "Exiting"
  EXIT
FI
echo "Cleaning update directory..."
rm -r -q /${TVOL}
echo "Creating directory structure..."
md /${TVOL}
md /${TVOL}/bin
md /${TVOL}/drv
#md /${TVOL}/etc
md /${TVOL}/inc
md /${TVOL}/include
md /${TVOL}/include/sys
md /${TVOL}/lib
md /${TVOL}/root
md /${TVOL}/root/acostest
md /${TVOL}/root/ftest
md /${TVOL}/root/shtest
md /${TVOL}/root/cshtest
md /${TVOL}/sbin
md /${TVOL}/sys
md /${TVOL}/sys/pm
md /${TVOL}/tmp
md /${TVOL}/usr
md /${TVOL}/usr/share
md /${TVOL}/usr/share/adm
md /${TVOL}/usr/share/extras
md /${TVOL}/usr/share/help
md /${TVOL}/usr/share/help/adm.help
md /${TVOL}/usr/share/help/sh.help
md /${TVOL}/usr/share/man
md /${TVOL}/var
md /${TVOL}/var/bbs
md /${TVOL}/var/log
md /${TVOL}/var/www

ECHO "Copying Files..."
FOR FILE IN (updatefiles)
  if [ -f ${SRC}/${FILE} ]
    cp -q ${SRC}/${FILE} /${TVOL}/${FILE}
  else
    echo "NOT FOUND: ${SRC}/${FILE}"
  fi
NEXT

cp -q ${SRC}/BASIC.SYSTEM /${TVOL}/bin/basic
cp -Y -Q ${SRC}/PRODOS.FX /${TVOL}/ProDOS

cp -q -r -y /MAKE/usr/share/adm/adm /${TVOL}/bin/adm
cp -q -r -y /MAKE/usr/share/adm/help /${TVOL}/bin/help
cp -q -r -y /MAKE/usr/share/adm/update /${TVOL}/usr/share/adm/update
cp -q -r -y /MAKE/usr/share/help /${TVOL}/usr/share
cp -q -r -y /MAKE/usr/share/man /${TVOL}/usr/share

cd /${TVOL}
pak update.${B} -r *
ECHO "Update Created!"
MAN
TEXT /MAKE/usr/share/make/makeupdate
