#!/bin/sh
#
#  This script copies the latest BUILD (A2OSX.BUILD) to MAKE (A2OSX.MAKE)
#  and then verifies/corrects that files are in the right locations to 
#  be used by other MAKE processes such as those for making disk images
#
. makefuncs
CALL CS
CALL TBOX "    New BUILD File Copier"
#  Check for BUILD and MAKE online
IF [ -D "/A2OSX.BUILD" ]
ELSE
  ECHO "A2OSX.BUILD is not online or available."
  ECHO
  ECHO "Exiting"
  EXIT
FI
IF [ -D "/MAKE" ]
ELSE
  ECHO "A2OSX.MAKE is not online or available."
  ECHO
  ECHO "Exiting"
  EXIT
FI
CALL PRINTXY 8 0 "BUILD and MAKE volumes found."
ECHO -e "\n"
switch $#
  case 0
    read -p "Which Build No:  " B
    if [ -z $B ]
      echo -e "Error: Empty response\n\n"
      exit
    fi
	break
  case 1
    set B = $1
    if [ -i $B ]
    else
      echo -e "\n\n Error: Non-Numeric Parameter \n\n"
	  exit
    fi
	break
  default
    echo -e "\n\n Error: Too Many Parameters \n\n"
	exit
end
set TBLD = "/MAKE/BUILDS/BUILD${B}"
IF [ -D ${TBLD} ]
  ECHO -e "\n\nFound Existing BUILD ${B}\n"
  ECHO
  ECHO "If you want to overwrite an existing build, use 'clobber' utility."
  ECHO -e "\nExiting"
  EXIT
ELSE
  ECHO -e "\n\nNo BUILD ${B} found."
  ECHO
  ECHO "Creating Directory"
FI

if ![ -D /MAKE/BUILDS ]
  md /MAKE/BUILDS
fi

# this is where we take stuff from the BUILD media
md ${TBLD}
echo -e "\nCopying Files..."
cp -Y -Q -R /A2OSX.BUILD/* ${TBLD}
echo -e "\nBUILD $B Copied!"

# this is where we take stuff not found in BUILD media and install it
echo "Installing non-build files into target build..."
if ![ -d ${TBLD}/var ] ; md ${TBLD}/var ; fi
if ![ -d ${TBLD}/var/bbs ] ; md ${TBLD}/var/bbs ; fi
if ![ -d ${TBLD}/var/www ] ; md ${TBLD}/var/www ; fi

cp -Q -R /MAKE/var/bbs/* ${TBLD}/var/bbs
cp -Q -R /MAKE/var/www/* ${TBLD}/var/www
################################## HERE!!!!!!!!!!!!!!!!!!!!1

for f in "httpd.conf bbsd.conf"
# maybe at some point we'll detect differences
#    echo "Checking ${f}"
#    cmp /MAKE/usr/share/etc/${f} ${TBLD}/etc/${f}
    cp -q -y /MAKE/usr/share/etc/${f} ${TBLD}/etc/${f}
next


# compress ProDOS binaries
cd ${TBLD}
echo -e "\nPacking PRODOS.FX..."
pakme PRODOS PRODOS.FX.PAKED
# TODO: redundant copy of PRODOS and PRODOS.FX, needs to be fixed in other makemedia scripts
cp PRODOS PRODOS.FX
cp -q -y PRODOS* /MAKE
cd sys

# no more km
#if ![ -D km ]
#    md km
#fi
#cp km* km
#rm -q km*

echo "=== Copying PM"
if ![ -D pm ]
  md pm
fi
cp -Y pm* pm

#cp pm/pm.ramworks

#echo "=== Making BLEED"
# TODO: reename pak
pakme kernel kernel.PAKED
#echo "Fixed Up Files to make images"
#ECHO "PREFIX BUILDS/BUILD${B}" > /MAKE/BLEED
#ECHO "-A2OSX.SYSTEM" >> /MAKE/BLEED
#ECHO -e "BLEED file updated!  Process Complete.  Exiting.\n"
MAN
TEXT /MAKE/usr/share/make/copybuild
