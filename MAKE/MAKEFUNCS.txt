# Title Box function
# CALL TBOX Str_Title
function TBOX {
  echo -e -N "\e[1;20H"
  echo -e "\e[7m                                        \e[0m"
  set LINE = 2
  while [ $LINE -NE 5 ]
    echo -e -N "\e[${LINE};20H"
    echo -e "\e[7m \e[0m                                      \e[7m \e[0m"
    set LINE = $LINE + 1
  loop
  echo -e -N "\e[5;20H"
  echo -e "\e[7m                                        \e[0m"
  echo -e -N "\e[3;22H $1"
  call PRINTXY 7 0 ""
}

# return a temporary filename in $TEMP
function mktemp {
    set TBASE = "${BOOT}tmp"
    if [ -D /RAM5 ]
        set TBASE = "/RAM5/tmp"
    fi
    mkdir -p ${TBASE} > /dev/null
    set TEMP = "${TBASE}t$$.tmp"
}


# Find slot/drive for volume name specified
# call findSlotDrive volName expectedSlDr
# returns slot drive in $SLDR and $A is "" if found and volname@sldr if not
function findSlotDrive {
    # should be bin/mktemp?
    call mktemp
    ls -l / > ${TEMP}0
    grep "/$1" ${TEMP}0 | cut -m 40 -n 43 > ${TEMP}1
    set A = `stat -S ${TEMP}1`
    if ![ $A = 0 ]
        set SLDR = `cat ${TEMP}1`
        set A = ""
    else
        set SLDR = $2
        grep "$2" ${TEMP}0 | cut -m 65 > ${TEMP}1
        set A = `cat ${TEMP}1`
    fi
    rm -q -c ${TEMP}*
    set TEMP =
}


# gets build command line args
function getArgs {
    while [ $# -gt 0 ] and ![ -z $1 ]
        if [ $# -eq 0 ]
            break
        fi

        # this line has been breaking TODO BUG
        if [ $1 = "-h" ] or [ $1 = "-H" ]
            echo "Options:"
            echo "    -y        - proceed 'yes' if default target volume name found"
            echo "    -b type   - build type: R)elease, C)andidate, S)table"
            shift
        fi

        if [ $1 = "-y" ] or [ $1 = "-Y" ]
            set ALL_YES = 1
            shift
        fi

        if [ $1 = "-b" ] or [ $1 = "-B" ]
            shift
            set RCS = $1
            shift
        fi

        if [ $1 -gt 0 ]
            set B = $1
            shift
        fi
    loop

    echo "AFTER: ALL_YES=${ALL_YES}  RCS=${RCS}   B=${B}"
}
