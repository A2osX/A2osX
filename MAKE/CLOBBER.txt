NEW
  PREFIX
AUTO 4,1
#!/bin/sh
#
# CLOBBER
#
# tool to list what builds are on the /MAKE volume and also easily
# remove older ones.
#
. makefuncs
call CS
call TBOX "    CLOBBER BUILD"
#  Check for MAKE online
if [ -D "/MAKE" ]
else
  echo "A2OSX.MAKE is not online or available."
  echo
  echo "Exiting"
  exit
fi
call PRINTXY 8 0 "MAKE volume found."
echo -e "\n"
switch $#
  case 0
    ls /MAKE/BUILDS
    echo -e "\n"
    read -p "Which Build No:  " B
    if [ -z $B ]
      echo -e "Error: Empty response\n\n"
      exit
    fi
	break
  case 1
    set B = $1
    if [ -i $B ]
    else
      echo -e "\n\n Error: Non-Numeric Parameter \n\n"
	  exit
    fi
	break
  default
    echo -e "\n\n Error: Too Many Parameters \n\n"
	exit
end
set SRC = "/MAKE/BUILDS/BUILD${B}"
if ![ -D ${SRC} ]
  echo -e "\n\nDid not find existing BUILD ${B}\n"
  echo
  echo "Exiting"
  exit
fi

if ![ -D /MAKE/BUILDS ]
  md /MAKE/BUILDS
fi

pushd /MAKE/BUILDS
rm -r -c BUILD${B}
popd

MAN
TEXT /MAKE/usr/share/make/clobber
